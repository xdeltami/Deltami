   169→   169→local function updateTilePrompt(tile)
   170→   170→        local prompt = tile.prompt
   171→   171→        if not prompt then
   172→   172→                return
   173→   173→        end
   174→   174→        local plot = plots[tile.plotIndex]
   175→   175→        local owner = plot and plot.owner
   176→   176→
   177→   177→        if not owner then
   178→   178→                prompt.Enabled = false
   179→   179→                return
   180→   180→        end
   181→   181→
   182→   182→        if tile.state == "empty" then
   183→   183→                prompt.ActionText = "Tanam Bibit"
   184→   184→                prompt.ObjectText = "Petak Ladang"
   185→   185→                prompt.HoldDuration = 0
   186→   186→                prompt.Enabled = true
   187→   187→        elseif tile.state == "growing" then
   188→   188→                local data = DataService.get(owner)
   189→   189→                local canWater = data.Tools.can and not tile.watered
   190→   190→                if canWater then
   191→   191→                        prompt.ActionText = "Siram"
   192→   192→                        prompt.ObjectText = tile.seed .. " tumbuh"
   193→   193→                        prompt.HoldDuration = 0.4
   194→   194→                        prompt.Enabled = true
   195→   195→                else
   196→   196→                        prompt.Enabled = false
   197→   197→                end
   198→   198→        elseif tile.state == "ready" then
   199→   199→                prompt.ActionText = "Panen"
   200→   200→                prompt.ObjectText = cropById[tile.seed] and cropById[tile.seed].name or "Hasil"
   201→   201→                prompt.HoldDuration = 0
   202→   202→                prompt.Enabled = true
   203→   203→        end
   204→   204→end
   205→   205→
   206→   206→local function onTileTriggered(player: Player, tile)
   207→   207→        local plot = plots[tile.plotIndex]
   208→   208→        if not plot or plot.owner ~= player or not DataService.has(player) then
   209→   209→                return
   210→   210→        end
   211→   211→
   212→   212→        if tile.state == "empty" then
   213→   213→                Remotes.get("OpenPlant"):FireClient(player, tile.plotIndex, tile.tileIndex)
   214→   214→        elseif tile.state == "growing" then
   215→   215→                local data = DataService.get(player)
   216→   216→                if not data.Tools.can then
   217→   217→                        Remotes.get("Notify"):FireClient(player, "Butuh Alat", "Kamu butuh Kaleng Siram - beli di Toko Alat.")
   218→   218→                        return
   219→   219→                end
   220→   220→                if tile.watered then
   221→   221→                        return
   222→   222→                end
   223→   223→                tile.watered = true
   224→   224→                tile.plantedAt -= tile.growTime * Config.Economy.WaterBoost
   225→   225→                Remotes.get("Notify"):FireClient(player, "Disiram!", "Tanamanmu tumbuh 25% lebih cepat.")
   226→   226→                DataService.addXP(player, 2)
   227→   227→                DataService.push(player)
   228→   228→        elseif tile.state == "ready" then
   229→   229→                local crop = cropById[tile.seed]
   230→   230→                if not crop then
   231→   231→                        return
   232→   232→                end
   233→   233→                -- v20: percikan panen (sekali pakai, ringan untuk mobile)
   234→   234→                Util.burst(tile.part, Color3.fromRGB(crop.color[1], crop.color[2], crop.color[3]), 10, 6)
   235→   235→                DataService.giveItem(player, "crop", crop.id, 1)
   236→   236→                DataService.addXP(player, 4)
   237→   237→                QuestService.progress(player, "harvest", 1)
   238→   238→                tile.state = "empty"
   239→   239→                tile.seed = nil
   240→   240→                tile.plantedAt = 0
   241→   241→                tile.growTime = 0
   242→   242→                tile.watered = false
   243→   243→                tile.bucket = -1
   244→   244→                destroyPlant(tile)
   245→   245→                updateTilePrompt(tile)
   246→   246→                DataService.push(player)
   247→   247→        end
   248→   248→end

   258→   258→local function updateSign(plot)

   266→   266→        else
   267→   267→                local price = priceOfPlot(plot.index)
   268→   268→                signText.Text = price <= 0
   269→   269→                        and "LADANG " .. plot.index .. " - AMBIL ALIH (GRATIS)"

   275→   275→local function onClaimTriggered(player: Player, plot)
   276→   276→        if plot.owner then
   277→   277→                if plot.owner == player then
   278→   278→                        Remotes.get("Notify"):FireClient(player, "Ladangmu", "Ini sudah ladang milikmu. Tanami bibit di petak!")
   279→   279→                end
   280→   280→                return
   281→   281→        end
   282→   282→        -- v4 EKONOMI SYARIAH: ladang 1 GRATIS; ladang lain DIBELI dengan Micda
   283→   283→        -- (bayar penuh di muka - tanpa cicilan, tanpa bunga, tanpa riba)
   284→   284→        local price = priceOfPlot(plot.index)
   285→   285→        if price > 0 then
   286→   286→                local data = DataService.get(player)
   287→   287→                if data.Coins < price then
   288→   288→                        Remotes.get("Notify"):FireClient(player, "Micda Kurang",
   289→   289→                                "Ladang " .. plot.index .. " dijual " .. price .. " Micda (bayar penuh, tanpa bunga). Micdamu: " .. data.Coins .. ".")
   290→   290→                        return
   291→   291→                end
   292→   292→                DataService.addCoins(player, -price, false)
   293→   293→        end
   294→   294→        plot.owner = player
   295→   295→        updateSign(plot)
   296→   296→        for _, tile in plot.tiles do
   297→   297→                updateTilePrompt(tile)
   298→   298→        end
   299→   299→        local msg = price > 0
   300→   300→                and "Ladang " .. plot.index .. " kini milikmu (dibeli " .. price .. " Micda, lunas)! Beli bibit di Toko Bibit, lalu dekati petak tanah untuk menanam."
   301→   301→                or "Ladang " .. plot.index .. " kini milikmu (GRATIS)! Beli bibit di Toko Bibit (Pasar Timur), lalu dekati petak tanah untuk menanam."
   302→   302→        Remotes.get("Notify"):FireClient(player, "Ladang Baru!", msg)
   303→   303→        DataService.push(player)
   304→   304→end

   481→   481→-- TANAM (RemoteFunction dari panel pilih bibit)
   482→   482→--------------------------------------------------------------------
   483→   483→
   484→   484→local function onPlantSeed(player: Player, plotIndex: any, tileIndex: any, seedId: any)
   485→   485→        if not DataService.has(player) then
   486→   486→                return { ok = false, msg = "Data belum siap." }
   487→   487→        end
   488→   488→        if type(plotIndex) ~= "number" or type(tileIndex) ~= "number" or type(seedId) ~= "string" then
   489→   489→                return { ok = false, msg = "Permintaan tidak valid." }
   490→   490→        end
   491→   491→        local plot = plots[math.floor(plotIndex)]
   492→   492→        local tile = plot and plot.tiles[math.floor(tileIndex)]
   493→   493→        if not plot or not tile or plot.owner ~= player then
   494→   494→                return { ok = false, msg = "Petak tidak tersedia." }
   495→   495→        end
   496→   496→        if tile.state ~= "empty" then
   497→   497→                return { ok = false, msg = "Petak sudah ditanami." }
   498→   498→        end
   499→   499→        local crop = cropById[seedId]
   500→   500→        if not crop then
   501→   501→                return { ok = false, msg = "Bibit tidak dikenal." }
   502→   502→        end
   503→   503→        if DataService.levelOf(player) < crop.level then
   504→   504→                return { ok = false, msg = "Butuh level " .. crop.level .. " untuk bibit ini." }
   505→   505→        end
   506→   506→        if not DataService.takeItem(player, "seed", seedId, 1) then
   507→   507→                return { ok = false, msg = "Bibit habis. Beli lagi di Toko Bibit!" }
   508→   508→        end
   509→   509→
   510→   510→        tile.state = "growing"
   511→   511→        tile.seed = seedId
   512→   512→        tile.plantedAt = time()
   513→   513→        tile.growTime = crop.grow
   514→   514→        tile.watered = false
   515→   515→        tile.bucket = 0
   516→   516→        buildPlantVisual(tile, 0)
   517→   517→        updateTilePrompt(tile)
   518→   518→        DataService.addXP(player, 2)
   519→   519→        DataService.push(player)
   520→   520→        return { ok = true, msg = crop.name .. " ditanam! Kembali setelah ±" .. crop.grow .. " detik." }
   521→   521→end

   524→   524→-- LOOP PERTUMBUHAN
   525→   525→--------------------------------------------------------------------
   526→   526→
   527→   527→function FarmService.startLoop()
   528→   528→        task.spawn(function()
   529→   529→                while true do
   530→   530→                        task.wait(2)
   531→   531→                        for _, plot in plots do
   532→   532→                                if not plot.owner then
   533→   533→                                        continue
   534→   534→                                end
   535→   535→                                for _, tile in plot.tiles do
   536→   536→                                        if tile.state ~= "growing" then
   537→   537→                                                continue
   538→   538→                                        end
   539→   539→                                        local progress = (time() - tile.plantedAt) / math.max(1, tile.growTime)
   540→   540→                                        if progress >= 1 then
   541→   541→                                                tile.state = "ready"
   542→   542→                                                buildPlantVisual(tile, 3)
   543→   543→                                                updateTilePrompt(tile)
   544→   544→                                        else
   545→   545→                                                local bucket = math.floor(progress * 3) -- 0..2
   546→   546→                                                if bucket ~= tile.bucket then
   547→   547→                                                        tile.bucket = bucket
   548→   548→                                                        buildPlantVisual(tile, bucket)
   549→   549→                                                end
   550→   550→                                                local prompt = tile.prompt
   551→   551→                                                if prompt and not prompt.Enabled then
   552→   552→                                                        updateTilePrompt(tile) -- misal pemain baru saja beli kaleng siram
   553→   553→                                                end
   554→   554→                                        end
   555→   555→                                end
   556→   556→                        end
   557→   557→                end
   558→   558→        end)
   559→   559→end

   564→   564→
   565→   565→function FarmService.getPlotState(player: Player)
   566→   566→        for _, plot in plots do
   567→   567→                if plot.owner == player then
   568→   568→                        local tilesOut = {}
   569→   569→                        for i, tile in plot.tiles do
   570→   570→                                local entry: any = { state = tile.state }
   571→   571→                                if tile.state ~= "empty" then
   572→   572→                                        entry.seed = tile.seed
   573→   573→                                        entry.watered = tile.watered
   574→   574→                                        if tile.state == "growing" then
   575→   575→                                                entry.remain = math.max(0, math.ceil(tile.growTime - (time() - tile.plantedAt)))
   576→   576→                                        end
   577→   577→                                end
   578→   578→                                tilesOut[i] = entry
   579→   579→                        end
   580→   580→                        return { claimed = true, index = plot.index, tiles = tilesOut }
   581→   581→                end
   582→   582→        end
   583→   583→        return { claimed = false, tiles = {} }
   584→   584→end

   586→   586→--- Pemain keluar: hasil siap panen otomatis masuk tas,
   587→   587→--- tanaman yang masih tumbuh hangus (ladang dibebaskan).
   588→   588→function FarmService.releasePlot(player: Player)
   589→   589→        for _, plot in plots do
   590→   590→                if plot.owner == player then
   591→   591→                        local harvested = 0
   592→   592→                        for _, tile in plot.tiles do
   593→   593→                                if tile.state == "ready" and tile.seed then
   594→   594→                                        DataService.giveItem(player, "crop", tile.seed, 1)
   595→   595→                                        harvested += 1
   596→   596→                                end
   597→   597→                                tile.state = "empty"
   598→   598→                                tile.seed = nil
   599→   599→                                tile.plantedAt = 0
   600→   600→                                tile.growTime = 0
   601→   601→                                tile.watered = false
   602→   602→                                tile.bucket = -1
   603→   603→                                destroyPlant(tile)
   604→   604→                                updateTilePrompt(tile)
   605→   605→                        end
   606→   606→                        if harvested > 0 then
   607→   607→                                Remotes.get("Notify"):FireClient(player, "Panen Otomatis",
   608→   608→                                        harvested .. " hasil siap panen otomatis masuk tas sebelum kamu pergi.")
   609→   609→                        end
   610→   610→                        plot.owner = nil
   611→   611→                        updateSign(plot)
   612→   612→                end
   613→   613→        end
   614→   614→end
