    33→local function fmt(n: number): string
    34→        local s = tostring(math.floor(n))
    35→        local rev = string.reverse(s)
    36→        rev = rev:gsub("(%d%d%d)", "%1.")
    37→        rev = rev:gsub("%.$", "")
    38→        return string.reverse(rev)
    39→end
