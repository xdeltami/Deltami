    55→                TextScaled = true,
    56→                AutoButtonColor = true,

    61→                if k ~= "parent" and k ~= "_radius" then

    63→                end
    64→        end

   274→function UIController:startTips()
   275→        task.spawn(function()
   276→                local idx = 0
   277→                while true do
   278→                        idx += 1
   279→                        local tip = TIPS[(idx - 1) % #TIPS + 1]

   282→                        end
   283→                        task.wait(14)
   284→                end

   335→function UIController:renderState()
   336→        local s = self.state
   337→        if not s then
   338→                return
   339→        end
   340→        self.coinText.Text = Util.formatNumber(s.Coins)
   341→        self.levelText.Text = "Level " .. s.Level .. (s.Dev and "  [DEV]" or "")
   342→        self.xpBar.Size = UDim2.fromScale(math.clamp(s.LevelPct, 0.02, 1), 1)
   343→
   344→        -- refresh panel terbuka supaya harga/inv tetap segar
   345→        if self.activePanel == "Shop" and self.shopId then
   346→                self:renderShop()
   347→        elseif self.activePanel == "Plant" and self.pendingPlant then
   348→                self:renderPlantPanel()
   349→        elseif self.activePanel == "Inv" then
   350→                self:renderInv()
   351→        end
   352→end

   475→function UIController:petBonus(): number
   476→        local s = self.state
   477→        if not s or s.ActivePet == "" then
   478→                return 0
   479→        end
   480→        for _, pet in Config.Pets do
   481→                if pet.id == s.ActivePet then
   482→                        return pet.bonus
   483→                end
   484→        end
   485→        return 0
   486→end

   488→function UIController:clientPrice(baseSell: number, itemId: string): number
   489→        local s = self.state
   490→        local demand = (s and s.Demand and s.Demand[itemId]) or 1
   491→        local price = baseSell * demand * (1 + self:petBonus())
   492→        return math.max(1, math.floor(price + 0.5))
   493→end

   495→function UIController:renderShop()
   496→        local s = self.state
   497→        local list = self.shopList
   498→        for _, child in list:GetChildren() do
   499→                if child:IsA("Frame") then
   500→                        child:Destroy()
   501→                end
   502→        end
   503→        if not s then
   504→                return
   505→        end
   506→
   507→        local titles = { bibit = "Toko Bibit - Kak Tani", alat = "Toko Alat - Pak Bang", pet = "Toko Peliharaan - Mbah Sekar", jual = "Pasar Pengepul - Bu Laris" }
   508→        local judul = self.shopFrame:FindFirstChild("Judul") :: TextLabel?
   509→        if judul then
   510→                judul.Text = titles[self.shopId] or "Toko"
   511→        end

   514→        self.tabJualBtn.Visible = self.shopId == "jual"
   515→        self.tabBeliBtn.BackgroundColor3 = self.shopTab == "beli" and T.Green or T.PanelLight
   516→        self.tabJualBtn.BackgroundColor3 = self.shopTab == "jual" and T.Green or T.PanelLight

   523→                        { kind = "bag", data = s.Bag, lookup = Config.Resources, label = "SUMBER DAYA" },
   524→                        { kind = "fish", data = s.Fish, lookup = Config.Fish, label = "IKAN" },
   525→                }
   526→                for _, section in sections do
   527→                        for id, count in section.data do
   528→                                if count <= 0 then
   529→                                        continue
   530→                                end
   531→                                local def = nil
   532→                                for _, d in section.lookup do

   534→                                                def = d
   535→                                                break
   536→                                        end
   537→                                end
   538→                                if not def then
   539→                                        continue
   540→                                end
   541→                                any = true
   542→                                local unit = self:clientPrice(def.sell, id)
   543→                                local r = row(list, 66)
   544→                                local color = (def.color and Color3.fromRGB(def.color[1], def.color[2], def.color[3])) or T.Blue
   545→                                swatch(r, color)

   549→                                        BackgroundTransparency = 1,
   550→                                        Font = Enum.Font.GothamBold,
   551→                                        TextScaled = true,
   552→                                        TextColor3 = T.Text,
   553→                                        TextXAlignment = Enum.TextXAlignment.Left,
   554→                                        Text = def.name .. " x" .. count .. "  (@" .. unit .. " koin)",
   555→                                }, r)
   556→                                Util.create("TextLabel", {
   557→                                        Position = UDim2.fromOffset(62, 36),
   558→                                        Size = UDim2.new(1, -240, 0, 22),
   559→                                        BackgroundTransparency = 1,
   560→                                        Font = Enum.Font.Gotham,
   561→                                        TextScaled = true,
   562→                                        TextColor3 = T.TextDim,
   563→                                        TextXAlignment = Enum.TextXAlignment.Left,
   564→                                        Text = section.label .. " - permintaan pasar x" .. ((s.Demand and s.Demand[id]) or 1),
   565→                                }, r)
   566→                                btn({
   567→                                        parent = r,
   568→                                        AnchorPoint = Vector2.new(1, 0),
   569→                                        Position = UDim2.new(1, -66, 0, 14),
   570→                                        Size = UDim2.fromOffset(56, 38),
   571→                                        BackgroundColor3 = T.Gold,
