    23→    23→local function fmt(n: number): string
    24→    24→        local s = tostring(math.floor(n))
    25→    25→        local rev = string.reverse(s)
    26→    26→        rev = rev:gsub("(%d%d%d)", "%1.")
    27→    27→        rev = rev:gsub("%.$", "")
    28→    28→        return string.reverse(rev)
