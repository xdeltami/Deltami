   265→function UIController:openStory(zoneIndex: number)
   266→        task.spawn(function()
   267→                local ok, story = pcall(function()
   268→                        return Remotes.get("RequestStory"):InvokeServer(zoneIndex)
   269→                end)
   270→                if not ok or not story then
   271→                        return
   272→                end
   273→                if story.locked then
   274→                        self:toast("Zona Terkunci", "Kuasai zona sebelumnya dulu, penjelajah!")
   275→                        return
   276→                end
   277→                self.story = { zone = zoneIndex, pages = story.pages, page = 1 }
   278→                self.dlgName.Text = story.npc
   279→                self.dialogFrame.Visible = true
   280→                self:showStoryPage()
   281→        end)
   282→end
   283→
   284→function UIController:showStoryPage()
   285→        local story = self.story
   286→        if not story then
   287→                return
   288→        end
   289→        self.dlgText.Text = story.pages[story.page]
   290→        self.dlgNext.Text = story.page >= #story.pages and "Mulai Tantangan!" or "Lanjut"
   291→end
   292→
   293→function UIController:advanceDialog()
   294→        local story = self.story
   295→        if not story then
   296→                return
   297→        end
   298→        if story.page < #story.pages then
   299→                story.page += 1
   300→                self:showStoryPage()
   301→        else
   302→                self.dialogFrame.Visible = false
   303→                self:startQuiz(story.zone)
   304→        end
   305→end

   384→function UIController:startQuiz(zoneIndex: number)
   385→        task.spawn(function()
   386→                local ok, q = pcall(function()
   387→                        return Remotes.get("StartQuiz"):InvokeServer(zoneIndex)
   388→                end)
   389→                if not ok or not q then
   390→                        return
   391→                end
   392→                if q.locked then
   393→                        self:toast("Zona Terkunci", "Kuasai zona sebelumnya dulu!")
   394→                        return
   395→                end
   396→                self.activeQuiz = { qid = q.qid, zone = zoneIndex }
   397→                local zone = Config.Zones[zoneIndex]
   398→                self.quizTitle.Text = string.upper(zone.name) .. "  -  TANTANGAN " .. self:zoneCorrect(zoneIndex) .. "/" .. self:masteryTarget()
   399→                self.quizQuestion.Text = q.t
   400→                self.quizFeedback.Text = ""
   401→                for i, btn in self.optionButtons do
   402→                        btn.Text = q.opsi[i]
   403→                        btn.BackgroundColor3 = T.PanelLight
   404→                end
   405→                self.quizFrame.Visible = true
   406→        end)
   407→end
   408→
   409→function UIController:answer(choice: number)
   410→        local quiz = self.activeQuiz
   411→        if not quiz then
   412→                return
   413→        end
   414→        task.spawn(function()
   415→                local ok, result = pcall(function()
   416→                        return Remotes.get("AnswerQuiz"):InvokeServer(quiz.qid, choice)
   417→                end)
   418→                if not ok or not result then
   419→                        return
   420→                end
   421→                if result.expired then
   422→                        self.quizFrame.Visible = false
   423→                        self.activeQuiz = nil
   424→                        self:toast("Soal kedaluwarsa", "Coba tantangan lagi ya!")
   425→                        return
   426→                end
   427→                if result.correct then
   428→                        self.sound.PlaybackSpeed = 1.1
   429→                        self.sound:Play()
   430→                        self.optionButtons[choice].BackgroundColor3 = T.Green
   431→                        self.quizFeedback.TextColor3 = T.Green
   432→                        if result.justMastered then
   433→                                self.quizFeedback.Text = result.feedback .. "  ZONA DIKUASAI!"
   434→                        else
   435→                                self.quizFeedback.Text = result.feedback
   436→                        end
   437→                else
   438→                        self.sound.PlaybackSpeed = 0.7
   439→                        self.sound:Play()
   440→                        self.quizFeedback.TextColor3 = T.TextDim
   441→                        self.quizFeedback.Text = result.feedback
   442→                end
   443→                task.delay(1.6, function()
   444→                        if self.quizFrame.Visible then
   445→                                self.activeQuiz = nil
   446→                                self.quizFrame.Visible = false
   447→                                -- soal berikutnya otomatis kalau masih di depan NPC
   448→                                if result.correct then
   449→                                        self:startQuiz(quiz.zone)
   450→                                end
   451→                        end
   452→                end)
   453→        end)
   454→end

   804→function UIController:detectZone()
   805→        local char = player.Character
   806→        local root = char and char:FindFirstChild("HumanoidRootPart")
   807→        if not root then
   808→                return
   809→        end
   810→        local z = root.Position.Z
   811→        local idx = math.floor((z - Config.World.StartZ + Config.World.ZoneSpacing / 2) / Config.World.ZoneSpacing) + 1
   812→        idx = math.clamp(idx, 1, #Config.Zones)
   813→        if idx ~= self.zoneIndex then
   814→                self.zoneIndex = idx
   815→                local zone = Config.Zones[idx]
   816→                local color = Util.zoneColor(zone.color)
   817→                self.zoneName.Text = zone.name
   818→                self.zoneName.TextColor3 = color
   819→                self.zoneSlogan.Text = zone.slogan
   820→        end
   821→        -- progres zona di chip
   822→        if self.state then
   823→                local zone = Config.Zones[self.zoneIndex]
   824→                local done = self.state.Correct[zone.id] or 0
   825→                local mastered = self.state.Mastered[zone.id]
   826→                local unlocked = self.zoneIndex <= (self.state.Unlocked or 1)
   827→                if not unlocked then
   828→                        self.zoneProgress.Text = "TERKUNCI"
   829→                        self.zoneProgress.TextColor3 = T.Red
   830→                elseif mastered then
   831→                        self.zoneProgress.Text = "DIKUASAI"
   832→                        self.zoneProgress.TextColor3 = T.Green
   833→                else
   834→                        self.zoneProgress.Text = done .. "/" .. self:masteryTarget()
   835→                        self.zoneProgress.TextColor3 = T.Gold
   836→                end
   837→        end
   838→end

   850→function UIController:wire()
   851→        Remotes.get("StateSync").OnClientEvent:Connect(function(state)
   852→                self:applyState(state)
   853→        end)
   854→
   855→        Remotes.get("Notify").OnClientEvent:Connect(function(title, message)
   856→                self:toast(title, message)
   857→        end)
   858→
   859→        Remotes.get("ZoneEntered").OnClientEvent:Connect(function(info)
   860→                if info.type == "npc" then
   861→                        self:openStory(info.zone)
   862→                elseif info.type == "board" then
   863→                        self:openBahasan(info.zone)
   864→                end
   865→        end)
   866→
   867→        task.spawn(function()
   868→                while true do
   869→                        task.wait(0.4)
   870→                        self:detectZone()
   871→                end
   872→        end)
   873→
   874→        task.spawn(function()
   875→                while true do
   876→                        task.wait(20)
   877→                        if self.lbOpen then
   878→                                self:refreshLB()
   879→                        end
   880→                end
   881→        end)
   882→end
   883→
   884→return UIController
