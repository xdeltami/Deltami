   102→   102→local function buildMonsterModel(def: any, position: Vector3, isBoss: boolean): (Model, Part)
   103→   103→        local model = Instance.new("Model")
   104→   104→        model.Name = isBoss and "AngkaRaja" or ("Angka_" .. tostring(def.body.digit or "1"))
   105→   105→
   106→   106→        local bodyColor = Color3.fromRGB(def.body.color[1], def.body.color[2], def.body.color[3])
   107→   107→        local auraColor = Color3.fromRGB(def.body.aura[1], def.body.aura[2], def.body.aura[3])
   108→   108→        local size = def.body.size
   109→   109→        local digit = tostring(def.body.digit or "1")
   110→   110→
   111→   111→        -- dimensi angka: tinggi = size, lebar = 62%, tebal balok = 18%
   112→   112→        local H = size
   113→   113→        local W = H * 0.62
   114→   114→        local t = math.max(0.5, H * 0.18)
   115→   115→        local depth = math.max(0.6, t * 0.9)
   116→   116→

   119→   119→                Name = "Badan",
   120→   120→                Size = Vector3.new(W + 2, H + 2, depth + 2),
   121→   121→                CFrame = CFrame.new(position + Vector3.new(0, H / 2, 0)),
   122→   122→                Transparency = 1,
   123→   123→                CanCollide = true,
   124→   124→        })
   125→   125→        root.Parent = model
   126→   126→        model.PrimaryPart = root
   127→   127→
   128→   128→        -- segmen angka + aura neon mengelilingi tiap balok
   129→   129→        local segs = DIGIT_SEGMENTS[digit] or "bc"
   130→   130→        for i = 1, #segs do
   131→   131→                local s = SEG[string.sub(segs, i, i)]
   132→   132→                local barSize = s.horiz and Vector3.new(W, t, depth) or Vector3.new(t, H / 2, depth)

   134→   134→                part({
   135→   135→                        Name = "Angka" .. i,
   136→   136→                        Size = barSize,
   137→   137→                        CFrame = root.CFrame * CFrame.new(offset),

   139→   139→                        Material = isBoss and Enum.Material.Neon or Enum.Material.SmoothPlastic,
   140→   140→                        CanCollide = false,
   141→   141→                }).Parent = model
   142→   142→                part({
   143→   143→                        Name = "Aura" .. i,
   144→   144→                        Size = s.horiz and Vector3.new(W + t * 1.2, t * 2.1, depth * 0.8) or Vector3.new(t * 2.1, H / 2 + t * 1.2, depth * 0.8),
   145→   145→                        CFrame = root.CFrame * CFrame.new(offset),

   148→   148→                        Transparency = 0.68,

   151→   151→        end
   152→   152→
   153→   153→        -- mata kecil di atas angka supaya terlihat hidup
   154→   154→        local eyeZ = -(depth / 2 + 0.05)
   155→   155→        local eyeSize = math.max(0.35, t * 0.5)

   170→   170→                Name = "CincinAura",

   180→   180→        local spark = Instance.new("ParticleEmitter")

   182→   182→        spark.Lifetime = NumberRange.new(0.5, 0.9)
   183→   183→        spark.Speed = NumberRange.new(0.6, 1.6)

   202→   202→        prompt.HoldDuration = 0
   203→   203→        prompt.RequiresLineOfSight = false

   303→   303→                lastHit = 0,

   319→   319→        end
   320→   320→

   322→   322→        return entity

   332→   332→local function monsterDie(entity: any, byPlayer: Player?)
   333→   333→        entity.alive = false
   334→   334→        local def = entity.def
   335→   335→
   336→   336→        -- animasi kematian sederhana: tenggelam & hilang
   337→   337→        for _, p in entity.model:GetChildren() do
   338→   338→                if p:IsA("BasePart") then
   339→   339→                        p.CanCollide = false
   340→   340→                        disablePrompt(p)
   341→   341→                end
   342→   342→        end
   343→   343→        local model = entity.model
   344→   344→        do
   345→   345→                -- ledakan aura saat tumbang
   346→   346→                local root = entity.body
   347→   347→                if root and root.Parent then
   348→   348→                        local burst = Instance.new("ParticleEmitter")
   349→   349→                        burst.Rate = 0
   350→   350→                        burst.Speed = NumberRange.new(8, 14)
   351→   351→                        burst.Lifetime = NumberRange.new(0.4, 0.8)
   352→   352→                        burst.SpreadAngle = Vector2.new(180, 180)
   353→   353→                        local c = Color3.fromRGB(def.body.aura[1], def.body.aura[2], def.body.aura[3])
   354→   354→                        burst.Color = ColorSequence.new(c)
   355→   355→                        burst.Size = NumberSequence.new(0.4)
   356→   356→                        burst.Parent = root
   357→   357→                        burst:Emit(20)
   358→   358→                end
   359→   359→        end
   360→   360→        task.spawn(function()
   361→   361→                for t = 1, 10 do
   362→   362→                        if model.Parent then
   363→   363→                                model:PivotTo(model:GetPivot() - Vector3.new(0, 0.25, 0))
   364→   364→                                for _, p in model:GetDescendants() do
   365→   365→                                        if p:IsA("BasePart") and p.Transparency < 1 then
   366→   366→                                                p.Transparency = math.min(1, t / 8)
   367→   367→                                        end
   368→   368→                                end
   369→   369→                        end
   370→   370→                        task.wait(0.05)
   371→   371→                end
   372→   372→                if entity.model.Parent then
   373→   373→                        entity.model:Destroy()
   374→   374→                end
   375→   375→        end)
   376→   376→
   377→   377→        monsters["m" .. entity.unique] = nil
   378→   378→
   379→   379→        if entity.isBoss then
   380→   380→                boss = nil
   381→   381→                return -- boss tidak respawn otomatis (diatur bossLoop)
   382→   382→        end
   383→   383→
   384→   384→        -- respawn di zona setelah jeda
   385→   385→        task.delay(def.respawn, function()
   386→   386→                if monsterFolder and monsterFolder.Parent then
   387→   387→                        spawnMonster(def, entity.zone)
   388→   388→                end
   389→   389→        end)
   390→   390→end

   440→   440→local function findNearestMonster(player: Player, maxDist: number)
   441→   441→        local root = player.Character and player.Character.PrimaryPart
   442→   442→        if not root then
   443→   443→                return nil
   444→   444→        end
   445→   445→        local best, bestDist = nil, maxDist
   446→   446→        for _, e in monsters do
   447→   447→                if e.alive and e.body.Parent then
   448→   448→                        local d = (e.body.Position - root.Position).Magnitude
   449→   449→                        if d < bestDist then
   450→   450→                                best, bestDist = e, d
   451→   451→                        end

   457→   457→local function hitEntity(player: Player, entity: any, mult: number, stats: any, level: number): number
   458→   458→        local atk = Balance.playerAttack({
   459→   459→                weaponAtk = EquipService.weaponAtk(player),
   460→   460→                str = stats.str,
   461→   461→                dex = stats.dex,
   462→   462→                int = stats.int,
   463→   463→                level = level,
   464→   464→        })
   465→   465→        atk = math.floor(atk * mult)
   466→   466→        if rng:NextNumber() < Balance.critChance(stats.dex) then
   467→   467→                atk = math.floor(atk * Balance.CritMult + 0.5)
   468→   468→        end
   469→   469→        local def = entity.def.def or 0
   470→   470→        local dmg = math.max(1, math.floor(atk * (1 - def / (def + 60)) + 0.5))
   471→   471→        entity.hp -= dmg
   472→   472→        local bb = entity.bb
   473→   473→        bb.bar.Size = UDim2.fromScale(math.clamp(entity.hp / entity.maxHp, 0, 1), 1)
   474→   474→        bb.label.Text = math.max(0, entity.hp) .. " / " .. entity.maxHp
   475→   475→        if entity.isBoss then
   476→   476→                bossCredits[player] = (bossCredits[player] or 0) + dmg
   477→   477→        end
   478→   478→        local killed = entity.hp <= 0
   479→   479→        damageTag(entity, dmg, mult > 1.6, killed)
   480→   480→        if killed then
   481→   481→                CombatService.onKill(player, entity)
   482→   482→                if entity.isBoss then
   483→   483→                        CombatService.finishBoss()
   484→   484→                end
   485→   485→                monsterDie(entity, player)
   486→   486→        else
   487→   487→                entity.target = player
   488→   488→        end
   489→   489→        return dmg
   490→   490→end

   492→   492→function CombatService.useSkill(player: Player, payload: any)
   493→   493→        if not DataService.has(player) or type(payload) ~= "table" then
   494→   494→                return { ok = false, msg = "Tidak valid." }
   495→   495→        end
   496→   496→        if not isAliveCharacter(player) then
   497→   497→                return { ok = false, msg = "Kamu tidak bisa bertarung sekarang." }
   498→   498→        end
   499→   499→        local skill = nil
   500→   500→        for _, s in ipairs(Balance.Skills) do
   501→   501→                if s.id == payload.skillId then
   502→   502→                        skill = s
   503→   503→                        break
   504→   504→                end
   505→   505→        end
   506→   506→        if not skill then
   507→   507→                return { ok = false, msg = "Skill tidak dikenal." }
   508→   508→        end
   509→   509→        if DataService.levelOf(player) < skill.level then
   510→   510→                return { ok = false, msg = "Skill terbuka di level " .. skill.level .. "." }
   511→   511→        end
   512→   512→        if EquipService.weaponAtk(player) <= 0 then
   513→   513→                return { ok = false, msg = skill.name .. " butuh PEDANG - tempa dulu di Pak Tempa!" }
   514→   514→        end
   515→   515→        local now = os.clock()
   516→   516→        lastSkill[player] = lastSkill[player] or {}
   517→   517→        local readyAt = lastSkill[player][skill.id] or 0
   518→   518→        if now < readyAt then
   519→   519→                return { ok = false, msg = skill.name .. " belum siap (" .. math.ceil(readyAt - now) .. "s).", cd = math.ceil(readyAt - now) }
   520→   520→        end
   521→   521→        local stats = statsOf(player)
   522→   522→        local level = DataService.levelOf(player)
   523→   523→        local char = player.Character
   524→   524→        if not char or not char.PrimaryPart then
   525→   525→                return { ok = false, msg = "Karakter belum siap." }
   526→   526→        end
   527→   527→
   528→   528→        if skill.aoe then
   529→   529→                local root = char.PrimaryPart
   530→   530→                local hits = 0
   531→   531→                local total = 0
   532→   532→                for _, e in monsters do
   533→   533→                        if e.alive and e.body.Parent and (e.body.Position - root.Position).Magnitude <= 16 then
   534→   534→                                total += hitEntity(player, e, skill.mult, stats, level)
   535→   535→                                hits += 1
   536→   536→                        end
   537→   537→                end
   538→   538→                if hits == 0 then
   539→   539→                        return { ok = false, msg = "Tidak ada angka di sekitarmu." }
   540→   540→                end
   541→   541→                lastSkill[player][skill.id] = now + skill.cooldown
   542→   542→                playSound(char.PrimaryPart, "swordlunge.wav", 1, 0.6)
   543→   543→                DataService.push(player)
   544→   544→                return { ok = true, msg = skill.name .. " menghantam " .. hits .. " angka! Total " .. total .. " damage." }
   545→   545→        end
   546→   546→
   547→   547→        local target = findNearestMonster(player, 16)
   548→   548→        if not target then
   549→   549→                return { ok = false, msg = "Tidak ada angka di dekatmu." }
   550→   550→        end
   551→   551→        local dmg = hitEntity(player, target, skill.mult, stats, level)
   552→   552→        lastSkill[player][skill.id] = now + skill.cooldown
   553→   553→        playSound(char.PrimaryPart, "swordlunge.wav", 1.1, 0.55)
   554→   554→        DataService.push(player)
   555→   555→        return { ok = true, msg = skill.name .. "! " .. dmg .. " damage ke " .. target.def.name .. "." }
   556→   556→end

   566→   566→function CombatService.playerAttack(player: Player, entity: any)
   567→   567→        if not entity or not entity.alive or not DataService.has(player) then
   568→   568→                return { ok = false, msg = "Target tidak valid." }
   569→   569→        end
   570→   570→        if not isAliveCharacter(player) then
   571→   571→                return { ok = false, msg = "Kamu tidak bisa menyerang sekarang." }
   572→   572→        end
   573→   573→        -- jarak server-side
   574→   574→        local root = player.Character and player.Character.PrimaryPart
   575→   575→        if not root or (root.Position - entity.body.Position).Magnitude > 14 then
   576→   576→                return { ok = false, msg = "Terlalu jauh!" }
   577→   577→        end
   578→   578→        -- jeda serangan (AGI mempercepat)
   579→   579→        local now = os.clock()
   580→   580→        local cd = Balance.attackCooldown(statsOf(player).agi)
   581→   581→        if lastAttack[player] and now - lastAttack[player] < cd then
   582→   582→                return { ok = false, msg = "Serangan belum siap." }
   583→   583→        end
   584→   584→        lastAttack[player] = now
   585→   585→
   586→   586→        local stats = statsOf(player)
   587→   587→        local level = DataService.levelOf(player)
   588→   588→        local atk = Balance.playerAttack({
   589→   589→                weaponAtk = EquipService.weaponAtk(player),
   590→   590→                str = stats.str,
   591→   591→                dex = stats.dex,
   592→   592→                int = stats.int,
   593→   593→                level = level,
   594→   594→        })
   595→   595→        local crit = rng:NextNumber() < Balance.critChance(stats.dex)
   596→   596→        if crit then
   597→   597→                atk = math.floor(atk * Balance.CritMult + 0.5)
   598→   598→        end
   599→   599→        -- kurva pelunakan dengan pertahanan monster
   600→   600→        local def = entity.def.def or 0
   601→   601→        local dmg = math.max(1, math.floor(atk * (1 - def / (def + 60)) + 0.5))
   602→   602→
   603→   603→        entity.hp -= dmg
   604→   604→        local bb = entity.bb
   605→   605→        bb.bar.Size = UDim2.fromScale(math.clamp(entity.hp / entity.maxHp, 0, 1), 1)
   606→   606→        bb.label.Text = math.max(0, entity.hp) .. " / " .. entity.maxHp
   607→   607→
   608→   608→        -- krediting boss raid
   609→   609→        if entity.isBoss then
   610→   610→                bossCredits[player] = (bossCredits[player] or 0) + dmg
   611→   611→        end
   612→   612→
   613→   613→        local killed = entity.hp <= 0
   614→   614→        damageTag(entity, dmg, crit, killed)
   615→   615→
   616→   616→        -- recoil visual: angka tersentak sedikit (juice)
   617→   617→        do
   618→   618→                local rootPart = entity.body
   619→   619→                if rootPart and rootPart.Parent and not killed then
   620→   620→                        local dir = (rootPart.Position - root.Position)
   621→   621→                        dir = Vector3.new(dir.X, 0, dir.Z)
   622→   622→                        if dir.Magnitude > 0.1 then
   623→   623→                                dir = dir.Unit
   624→   624→                        else
   625→   625→                                dir = root.CFrame.LookVector * -1
   626→   626→                        end
   627→   627→                        local base = rootPart.Position
   628→   628→                        entity.model:PivotTo(CFrame.new(base + dir * 1.1))
   629→   629→                        task.delay(0.12, function()
   630→   630→                                if entity.alive and entity.model.Parent and entity.body.Parent then
   631→   631→                                        entity.model:PivotTo(CFrame.new(base))
   632→   632→                                end
   633→   633→                        end)
   634→   634→                end
   635→   635→        end
   636→   636→
   637→   637→        if killed then
   638→   638→                CombatService.onKill(player, entity)
   639→   639→                if entity.isBoss then
   640→   640→                        -- bagi hadiah raid SEBELUM boss dibersihkan
   641→   641→                        CombatService.finishBoss()
   642→   642→                end
   643→   643→                monsterDie(entity, player)
   644→   644→        else
   645→   645→                -- monster berbalik mengejar penyerang
   646→   646→                entity.target = player
   647→   647→        end
   648→   648→
   649→   649→        DataService.push(player)
   650→   650→        return { ok = true, dmg = dmg, crit = crit, killed = killed, name = entity.def.name }
   651→   651→end

   699→   699→local function stepMonster(entity: any, dt: number)
   700→   700→        if not entity.alive or not entity.body.Parent then
   701→   701→                return
   702→   702→        end
   703→   703→        local def = entity.def
   704→   704→
   705→   705→        -- validasi target
   706→   706→        local targetPlayer = entity.target
   707→   707→        if targetPlayer then
   708→   708→                local valid = targetPlayer.Parent ~= nil and isAliveCharacter(targetPlayer)
   709→   709→                if valid then
   710→   710→                        local root = targetPlayer.Character.PrimaryPart
   711→   711→                        local dist = (root.Position - entity.home).Magnitude
   712→   712→                        if dist > def.aggro * 2.2 then
   713→   713→                                valid = false -- terlalu jauh dari rumahnya, pulang
   714→   714→                        end
   715→   715→                end
   716→   716→                if not valid then
   717→   717→                        entity.target = nil
   718→   718→                end
   719→   719→        end
   720→   720→
   721→   721→        -- cari target baru dalam radius aggro
   722→   722→        if not entity.target then
   723→   723→                for _, player in Players:GetPlayers() do
   724→   724→                        if isAliveCharacter(player) and DataService.has(player) then
   725→   725→                                local root = player.Character.PrimaryPart
   726→   726→                                if (root.Position - entity.body.Position).Magnitude <= def.aggro then
   727→   727→                                        entity.target = player
   728→   728→                                        break
   729→   729→                                end
   730→   730→                        end
   731→   731→                end
   732→   732→        end
   733→   733→
   734→   734→        local bodyPos = entity.body.Position
   735→   735→        local goal: Vector3
   736→   736→        local speed = def.speed
   737→   737→
   738→   738→        if entity.target then
   739→   739→                local root = entity.target.Character.PrimaryPart
   740→   740→                local dist = (root.Position - bodyPos).Magnitude
   741→   741→                if dist > 5 then
   742→   742→                        goal = root.Position
   743→   743→                else
   744→   744→                        goal = bodyPos -- dekat: berhenti & serang
   745→   745→                        local now = os.clock()
   746→   746→                        if now - entity.lastHit >= 1.5 then
   747→   747→                                entity.lastHit = now
   748→   748→                                -- damage ke pemain dengan pertahanan zirah + VIT
   749→   749→                                local pstats = statsOf(entity.target)
   750→   750→                                local defense = Balance.playerDefense(EquipService.armorDef(entity.target), pstats.vit)
   751→   751→                                local taken = Balance.damageTaken(def.atk, defense)
   752→   752→                                local humanoid = entity.target.Character:FindFirstChildOfClass("Humanoid")
   753→   753→                                if humanoid then
   754→   754→                                        humanoid:TakeDamage(taken)
   755→   755→                                end
   756→   756→                        end
   757→   757→                end
   758→   758→        else
   759→   759→                -- berkeliaran di sekitar rumah
   760→   760→                if (bodyPos - entity.wanderTo).Magnitude < 3 then
   761→   761→                        entity.wanderTo = entity.home + Vector3.new(rng:NextNumber(-12, 12), 0, rng:NextNumber(-12, 12))
   762→   762→                        return
   763→   763→                end
   764→   764→                goal = entity.wanderTo
   765→   765→                speed = def.speed * 0.45
   766→   766→        end
   767→   767→
   768→   768→        local dir = (goal - bodyPos) * Vector3.new(1, 0, 1)
   769→   769→        if dir.Magnitude > 0.4 then
   770→   770→                dir = dir.Unit
   771→   771→                local step = math.min(speed * dt, dir.Magnitude * 99)
   772→   772→                local newPos = bodyPos + dir * step
   773→   773→                -- v4: pindahkan SELURUH model (angka, aura, mata) supaya tidak
   774→   774→                -- ada part tertinggal berserakan (penyebab dunia terlihat kotor)
   775→   775→                entity.model:PivotTo(CFrame.lookAt(newPos, newPos + dir))
   776→   776→        end
   777→   777→end

   839→   839→function CombatService.finishBoss()
   840→   840→        if not boss then
   841→   841→                return

   843→   843→        local def = Balance.Boss
   844→   844→        local participants = {}

   853→   853→                return a.dmg > b.dmg

   857→   857→                local share = entry.dmg / math.max(1, totalDmg)

   859→   859→                local xp = math.max(def.participationXp, math.floor(def.xp * share))
   860→   860→                DataService.addCoins(entry.player, coin, true)
   861→   861→                DataService.addXP(entry.player, xp)
   862→   862→                local extra = ""
   863→   863→                if rank == 1 then
   864→   864→                        DataService.addCoins(entry.player, def.topBonusCoin, true)
   865→   865→                        for _, drop in ipairs(def.drops) do

   867→   867→                                DataService.giveItem(entry.player, "bag", drop.id, n)
   868→   868→                                extra = extra .. " +" .. n .. " " .. drop.id
   869→   869→                        end
   870→   870→                end
   871→   871→                notify(entry.player, "BOSS DIKALAHKAN!", rank == 1 and ("KAU PENYERANG TERAKHIR/TERBESAR! +" .. coin .. " Micda" .. extra)
   872→   872→                        or ("Kontribusi " .. math.floor(share * 100) .. "% - +" .. coin .. " Micda, +" .. xp .. " XP"))
   873→   873→                DataService.push(entry.player)
   874→   874→        end
   875→   875→        bossCredits = {}
   876→   876→end

   953→   953→        newFolder.Name = "AKM_Monster"
   954→   954→        newFolder.Parent = workspace
   955→   955→        monsterFolder = newFolder
   956→   956→
   957→   957→        Remotes.get("Attack").OnServerInvoke = function(player, monsterId)

   959→   959→                        return { ok = false, msg = "Tidak valid." }
   960→   960→                end
   961→   961→                local entity = monsters[monsterId] or boss
   962→   962→                if not entity or not entity.alive or ("m" .. entity.unique) ~= monsterId then
   963→   963→                        return { ok = false, msg = "Target tidak ada." }
   964→   964→                end
   965→   965→                return CombatService.playerAttack(player, entity)
   966→   966→        end

   968→   968→        Players.PlayerRemoving:Connect(function(player)
