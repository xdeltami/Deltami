   318→   318→local function dalamSkip(skips, v: number): boolean
   319→   319→        if not skips then
   320→   320→                return false
   321→   321→        end
   322→   322→        for _, r in ipairs(skips) do
   323→   323→                if v >= r[1] and v <= r[2] then
   324→   324→                        return true
   325→   325→                end
   326→   326→        end

   330→   330→-- Jalan aspal dua arah: slab aspal + marka putus-putus + trotoar timbul
   331→   331→-- dua sisi + jalur sepeda merah dua sisi. Sumbu jalan mengikuti sisi
   332→   332→-- terpanjang dari (ax,az)->(bx,bz).
   333→   333→local function jalanAspal(parent: Instance, ax: number, az: number, bx: number, bz: number, o)
   334→   334→        o = o or {}
   335→   335→        local lebar = o.lebar or 12
   336→   336→        local tebal = o.tebal or 0.2
   337→   337→        local horizontal = math.abs(bx - ax) >= math.abs(bz - az)
   338→   338→        local len = horizontal and (bx - ax) or (bz - az)
   339→   339→        local abslen = math.abs(len)
   340→   340→        local cx, cz = (ax + bx) / 2, (az + bz) / 2
   341→   341→        local m = Instance.new("Model")
   342→   342→        m.Name = "Jalan_" .. (o.nama or "Aspal")
   343→   343→        m.Parent = parent
   344→   344→        local rot = horizontal and 0 or math.rad(90)
   345→   345→        local cf0 = CFrame.new(cx, GT, cz) * CFrame.Angles(0, rot, 0)
   346→   346→        local aspal = part(m, Vector3.new(abslen, tebal, lebar), cf0 * CFrame.new(0, tebal / 2, 0), ASPAL, Enum.Material.Asphalt)
   347→   347→        aspal.Name = "JalanAspal"
   348→   348→        local nDash = math.floor((abslen - 6) / 7)
   349→   349→        for i = 0, nDash - 1 do
   350→   350→                local along = -abslen / 2 + 6 + i * 7
   351→   351→                if math.abs(along) < abslen / 2 - 2 then
   352→   352→                        local dash = part(m, Vector3.new(2.6, 0.06, 0.45), cf0 * CFrame.new(along, tebal + 0.07, 0), MARKA, Enum.Material.SmoothPlastic)
   353→   353→                        dash.Name = "MarkaPutih"
   354→   354→                        dash.CanCollide = false
   355→   355→                end
   356→   356→        end
   357→   357→        for _, s in ipairs({ -1, 1 }) do
   358→   358→                local trotoar = part(m, Vector3.new(abslen, 0.35, 2.4), cf0 * CFrame.new(0, 0.175, s * (lebar / 2 + 0.8)), TROTOAR, Enum.Material.Concrete)
   359→   359→                trotoar.Name = "Trotoar"
   360→   360→                local jalur = part(m, Vector3.new(abslen, 0.08, 2), cf0 * CFrame.new(0, tebal + 0.05, s * (lebar / 2 - 1.5)), SEPEDA, Enum.Material.SmoothPlastic)
   361→   361→                jalur.Name = "JalurSepeda"
   362→   362→                jalur.CanCollide = false
   363→   363→        end
   364→   364→        return m, horizontal, cx, cz, abslen
   365→   365→end

   367→   367→-- Lampu jalan + pohon pinggir jalan, bolak-balik sisi, dengan daftar
   368→   368→-- rentang koordinat yang dilewati (persimpangan/zebra/gerbang).
   369→   369→local function hiasJalan(parent: Instance, horizontal: boolean, cx: number, cz: number, abslen: number, lebar: number, skips)
   370→   370→        local s = -abslen / 2 + 14
   371→   371→        local k = 0
   372→   372→        while s <= abslen / 2 - 14 do
   373→   373→                local pusatSum = horizontal and cx or cz
   374→   374→                if not dalamSkip(skips, pusatSum + s) then
   375→   375→                        local side = (k % 2 == 0) and -1 or 1
   376→   376→                        local lx, lz
   377→   377→                        if horizontal then
   378→   378→                                lx, lz = cx + s, cz + side * (lebar / 2 + 1.2)
   379→   379→                        else
   380→   380→                                lx, lz = cx + side * (lebar / 2 + 1.2), cz + s
   381→   381→                        end
   382→   382→                        lampuTaman(parent, lx, lz)
   383→   383→                        local ta = s + 15
   384→   384→                        if not dalamSkip(skips, pusatSum + ta) then
   385→   385→                                local tx, tz
   386→   386→                                if horizontal then
   387→   387→                                        tx, tz = cx + ta, cz - side * (lebar / 2 + 2.8)
   388→   388→                                else
   389→   389→                                        tx, tz = cx - side * (lebar / 2 + 2.8), cz + ta
   390→   390→                                end
   391→   391→                                pohonAlam(parent, tx, tz, 0.72)
   392→   392→                        end
   393→   393→                end
   394→   394→                s += 30
   395→   395→                k += 1
   396→   396→        end
   397→   397→end
