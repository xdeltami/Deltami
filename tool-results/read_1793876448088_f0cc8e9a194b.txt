   102→local function buildMonsterModel(def: any, position: Vector3, isBoss: boolean): (Model, Part)
   103→        local model = Instance.new("Model")
   104→        model.Name = isBoss and "AngkaRaja" or ("Angka_" .. tostring(def.body.digit or "1"))
   105→
   106→        local bodyColor = Color3.fromRGB(def.body.color[1], def.body.color[2], def.body.color[3])
   107→        local auraColor = Color3.fromRGB(def.body.aura[1], def.body.aura[2], def.body.aura[3])
   108→        local size = def.body.size
   109→        local digit = tostring(def.body.digit or "1")
   110→
   111→        -- dimensi angka: tinggi = size, lebar = 62%, tebal balok = 18%
   112→        local H = size
   113→        local W = H * 0.62
   114→        local t = math.max(0.5, H * 0.18)
   115→        local depth = math.max(0.6, t * 0.9)
   116→
   117→        -- AKAR: part tak terlihat sebagai acuan gerak & penempel prompt/HP
   118→        local root = part({
   119→                Name = "Badan",
   120→                Size = Vector3.new(W + 2, H + 2, depth + 2),
   121→                CFrame = CFrame.new(position + Vector3.new(0, H / 2, 0)),
   122→                Transparency = 1,
   123→                CanCollide = true,
   124→        })
   125→        root.Parent = model
   126→        model.PrimaryPart = root
   127→
   128→        -- segmen angka + aura neon mengelilingi tiap balok
   129→        local segs = DIGIT_SEGMENTS[digit] or "bc"
   130→        for i = 1, #segs do
   131→                local s = SEG[string.sub(segs, i, i)]
   132→                local barSize = s.horiz and Vector3.new(W, t, depth) or Vector3.new(t, H / 2, depth)
   133→                local offset = Vector3.new(s.x * W, s.y * H, 0)
   134→                part({
   135→                        Name = "Angka" .. i,
   136→                        Size = barSize,
   137→                        CFrame = root.CFrame * CFrame.new(offset),
   138→                        Color = bodyColor,
   139→                        Material = isBoss and Enum.Material.Neon or Enum.Material.SmoothPlastic,
   140→                        CanCollide = false,
   141→                }).Parent = model
   142→                part({
   143→                        Name = "Aura" .. i,
   144→                        Size = s.horiz and Vector3.new(W + t * 1.2, t * 2.1, depth * 0.8) or Vector3.new(t * 2.1, H / 2 + t * 1.2, depth * 0.8),
   145→                        CFrame = root.CFrame * CFrame.new(offset),
   146→                        Color = auraColor,
   147→                        Material = Enum.Material.Neon,
   148→                        Transparency = 0.68,
   149→                        CanCollide = false,
   150→                }).Parent = model
   151→        end
   152→
   153→        -- mata kecil di atas angka supaya terlihat hidup
   154→        local eyeZ = -(depth / 2 + 0.05)
   155→        local eyeSize = math.max(0.35, t * 0.5)
   156→        for side = -1, 1, 2 do
   157→                part({
   158→                        Name = "Mata" .. side,
   159→                        Shape = Enum.PartType.Ball,
   160→                        Size = Vector3.new(1, 1, 1) * eyeSize,
   161→                        CFrame = root.CFrame * CFrame.new(Vector3.new(side * W * 0.18, H * 0.36, eyeZ)),
   162→                        Color = isBoss and Color3.fromRGB(255, 80, 60) or Color3.fromRGB(26, 26, 32),
   163→                        Material = Enum.Material.Neon,
   164→                        CanCollide = false,
   165→                }).Parent = model
   166→        end
   167→
   168→        -- cincin aura di tanah + percikan halus (ringan untuk mobile)
   169→        part({
   170→                Name = "CincinAura",
   171→                Shape = Enum.PartType.Cylinder,
   172→                Size = Vector3.new(0.18, W * 2.6, W * 2.6),
   173→                CFrame = CFrame.new(position + Vector3.new(0, 0.12, 0)) * CFrame.Angles(0, 0, math.rad(90)),
   174→                Color = auraColor,
   175→                Material = Enum.Material.Neon,
   176→                Transparency = 0.55,
   177→                CanCollide = false,
   178→        }).Parent = model

   180→        local spark = Instance.new("ParticleEmitter")
   181→        spark.Rate = isBoss and 8 or 3
   182→        spark.Lifetime = NumberRange.new(0.5, 0.9)
   183→        spark.Speed = NumberRange.new(0.6, 1.6)

   185→        spark.Color = ColorSequence.new(auraColor)
   186→        spark.Transparency = NumberSequence.new({ NumberSequenceKeypoint.new(0, 0.25), NumberSequenceKeypoint.new(1, 1) })
   187→        spark.Enabled = true
   188→        spark.Parent = root
   189→
   190→        if isBoss then
   191→                local light = Instance.new("PointLight")
   192→                light.Color = auraColor
   193→                light.Range = 30
   194→                light.Brightness = 1.4
   195→                light.Parent = root

   198→        local prompt = Instance.new("ProximityPrompt")
   199→        prompt.ActionText = "Serang"
   200→        prompt.ObjectText = def.name

   202→        prompt.HoldDuration = 0

   283→local function spawnMonster(def: any, zoneId: string, isBoss: boolean?)
   284→        local center = zoneCenter(zoneId)
   285→        local offset = Vector3.new(rng:NextNumber(-1, 1) * 12, 0, rng:NextNumber(-1, 1) * 12)
   286→        local position = center + offset
   287→        local model, body = buildMonsterModel(def, position, isBoss == true)
   288→
   289→        local maxHp = def.hp
   290→        local bb = hpBillboard(body, def.name, maxHp, maxHp, isBoss == true)
   291→
   292→        local entity = {
   293→                def = def,
   294→                zone = zoneId,
   295→                model = model,
   296→                body = body,
   297→                bb = bb,
   298→                hp = maxHp,
   299→                maxHp = maxHp,
   300→                alive = true,
   301→                home = position,
   302→                target = nil :: Player?,
   303→                lastHit = 0,
   304→                lastMove = 0,
   305→                wanderTo = position,
   306→                isBoss = isBoss == true,
   307→                unique = 0,
   308→        }
   309→        monsterSeq += 1
   310→        entity.unique = monsterSeq
   311→
   312→        local prompt = body:FindFirstChildOfClass("ProximityPrompt")
   313→        if prompt then
   314→                prompt.Triggered:Connect(function(player)
   315→                        if entity.alive then
   316→                                task.spawn(CombatService.playerAttack, player, entity)
   317→                        end
   318→                end)
   319→        end
   320→
   321→        monsters["m" .. entity.unique] = entity
   322→        return entity
   323→end

   332→local function monsterDie(entity: any, byPlayer: Player?)
   333→        entity.alive = false
   334→        local def = entity.def
   335→
   336→        -- animasi kematian sederhana: tenggelam & hilang
   337→        for _, p in entity.model:GetChildren() do
   338→                if p:IsA("BasePart") then
   339→                        p.CanCollide = false
   340→                        disablePrompt(p)
   341→                end
   342→        end
   343→        local model = entity.model
   344→        do
   345→                -- ledakan aura saat tumbang
   346→                local root = entity.body
   347→                if root and root.Parent then
   348→                        local burst = Instance.new("ParticleEmitter")
   349→                        burst.Rate = 0
   350→                        burst.Speed = NumberRange.new(8, 14)
   351→                        burst.Lifetime = NumberRange.new(0.4, 0.8)
   352→                        burst.SpreadAngle = Vector2.new(180, 180)
   353→                        local c = Color3.fromRGB(def.body.aura[1], def.body.aura[2], def.body.aura[3])
   354→                        burst.Color = ColorSequence.new(c)
   355→                        burst.Size = NumberSequence.new(0.4)
   356→                        burst.Parent = root
   357→                        burst:Emit(20)
   358→                end
   359→        end
   360→        task.spawn(function()
   361→                for t = 1, 10 do
   362→                        if model.Parent then
   363→                                model:PivotTo(model:GetPivot() - Vector3.new(0, 0.25, 0))
   364→                                for _, p in model:GetDescendants() do
   365→                                        if p:IsA("BasePart") and p.Transparency < 1 then
   366→                                                p.Transparency = math.min(1, t / 8)
   367→                                        end
   368→                                end
   369→                        end
   370→                        task.wait(0.05)
   371→                end
   372→                if entity.model.Parent then
   373→                        entity.model:Destroy()
   374→                end
   375→        end)
   376→
   377→        monsters["m" .. entity.unique] = nil
   378→
   379→        if entity.isBoss then
   380→                boss = nil
   381→                return -- boss tidak respawn otomatis (diatur bossLoop)
   382→        end
   383→
   384→        -- respawn di zona setelah jeda
   385→        task.delay(def.respawn, function()
   386→                if monsterFolder and monsterFolder.Parent then
   387→                        spawnMonster(def, entity.zone)
   388→                end
   389→        end)
   390→end

   440→local function findNearestMonster(player: Player, maxDist: number)
   441→        local root = player.Character and player.Character.PrimaryPart
   442→        if not root then
   443→                return nil
   444→        end
   445→        local best, bestDist = nil, maxDist
   446→        for _, e in monsters do
   447→                if e.alive and e.body.Parent then
   448→                        local d = (e.body.Position - root.Position).Magnitude
   449→                        if d < bestDist then
   450→                                best, bestDist = e, d
   451→                        end
   452→                end
   453→        end
   454→        return best
   455→end

   457→local function hitEntity(player: Player, entity: any, mult: number, stats: any, level: number): number
   458→        local atk = Balance.playerAttack({
   459→                weaponAtk = EquipService.weaponAtk(player),
   460→                str = stats.str,
   461→                dex = stats.dex,
   462→                int = stats.int,
   463→                level = level,
   464→        })
   465→        atk = math.floor(atk * mult)
   466→        if rng:NextNumber() < Balance.critChance(stats.dex) then
   467→                atk = math.floor(atk * Balance.CritMult + 0.5)
   468→        end
   469→        local def = entity.def.def or 0
   470→        local dmg = math.max(1, math.floor(atk * (1 - def / (def + 60)) + 0.5))
   471→        entity.hp -= dmg
   472→        local bb = entity.bb
   473→        bb.bar.Size = UDim2.fromScale(math.clamp(entity.hp / entity.maxHp, 0, 1), 1)
   474→        bb.label.Text = math.max(0, entity.hp) .. " / " .. entity.maxHp
   475→        if entity.isBoss then
   476→                bossCredits[player] = (bossCredits[player] or 0) + dmg
   477→        end
   478→        local killed = entity.hp <= 0
   479→        damageTag(entity, dmg, mult > 1.6, killed)
   480→        if killed then
   481→                CombatService.onKill(player, entity)
   482→                if entity.isBoss then
   483→                        CombatService.finishBoss()
   484→                end
   485→                monsterDie(entity, player)
   486→        else
   487→                entity.target = player
   488→        end
   489→        return dmg
   490→end

   492→function CombatService.useSkill(player: Player, payload: any)
   493→        if not DataService.has(player) or type(payload) ~= "table" then
   494→                return { ok = false, msg = "Tidak valid." }
   495→        end
   496→        if not isAliveCharacter(player) then
   497→                return { ok = false, msg = "Kamu tidak bisa bertarung sekarang." }
   498→        end
   499→        local skill = nil
   500→        for _, s in ipairs(Balance.Skills) do
   501→                if s.id == payload.skillId then
   502→                        skill = s
   503→                        break
   504→                end
   505→        end
   506→        if not skill then
   507→                return { ok = false, msg = "Skill tidak dikenal." }
   508→        end
   509→        if DataService.levelOf(player) < skill.level then
   510→                return { ok = false, msg = "Skill terbuka di level " .. skill.level .. "." }
   511→        end
   512→        if EquipService.weaponAtk(player) <= 0 then
   513→                return { ok = false, msg = skill.name .. " butuh PEDANG - tempa dulu di Pak Tempa!" }
   514→        end
   515→        local now = os.clock()
   516→        lastSkill[player] = lastSkill[player] or {}
   517→        local readyAt = lastSkill[player][skill.id] or 0
   518→        if now < readyAt then
   519→                return { ok = false, msg = skill.name .. " belum siap (" .. math.ceil(readyAt - now) .. "s).", cd = math.ceil(readyAt - now) }
   520→        end
   521→        local stats = statsOf(player)
   522→        local level = DataService.levelOf(player)
   523→        local char = player.Character
   524→        if not char or not char.PrimaryPart then
   525→                return { ok = false, msg = "Karakter belum siap." }
   526→        end
   527→
   528→        if skill.aoe then
   529→                local root = char.PrimaryPart
   530→                local hits = 0
   531→                local total = 0
   532→                for _, e in monsters do
   533→                        if e.alive and e.body.Parent and (e.body.Position - root.Position).Magnitude <= 16 then
   534→                                total += hitEntity(player, e, skill.mult, stats, level)
   535→                                hits += 1
   536→                        end
   537→                end
   538→                if hits == 0 then
   539→                        return { ok = false, msg = "Tidak ada angka di sekitarmu." }
   540→                end
   541→                lastSkill[player][skill.id] = now + skill.cooldown
   542→                playSound(char.PrimaryPart, "swordlunge.wav", 1, 0.6)
   543→                DataService.push(player)
   544→                return { ok = true, msg = skill.name .. " menghantam " .. hits .. " angka! Total " .. total .. " damage." }
   545→        end
   546→
   547→        local target = findNearestMonster(player, 16)
   548→        if not target then
   549→                return { ok = false, msg = "Tidak ada angka di dekatmu." }
   550→        end
   551→        local dmg = hitEntity(player, target, skill.mult, stats, level)
   552→        lastSkill[player][skill.id] = now + skill.cooldown
   553→        playSound(char.PrimaryPart, "swordlunge.wav", 1.1, 0.55)
   554→        DataService.push(player)
   555→        return { ok = true, msg = skill.name .. "! " .. dmg .. " damage ke " .. target.def.name .. "." }
   556→end

   559→        lastSkill[player] = nil

   566→function CombatService.playerAttack(player: Player, entity: any)
   567→        if not entity or not entity.alive or not DataService.has(player) then
   568→                return { ok = false, msg = "Target tidak valid." }
   569→        end
   570→        if not isAliveCharacter(player) then
   571→                return { ok = false, msg = "Kamu tidak bisa menyerang sekarang." }
   572→        end
   573→        -- jarak server-side
   574→        local root = player.Character and player.Character.PrimaryPart
   575→        if not root or (root.Position - entity.body.Position).Magnitude > 14 then
   576→                return { ok = false, msg = "Terlalu jauh!" }
   577→        end
   578→        -- jeda serangan (AGI mempercepat)
   579→        local now = os.clock()
   580→        local cd = Balance.attackCooldown(statsOf(player).agi)
   581→        if lastAttack[player] and now - lastAttack[player] < cd then
   582→                return { ok = false, msg = "Serangan belum siap." }
   583→        end
   584→        lastAttack[player] = now
   585→
   586→        local stats = statsOf(player)
   587→        local level = DataService.levelOf(player)
   588→        local atk = Balance.playerAttack({
   589→                weaponAtk = EquipService.weaponAtk(player),
   590→                str = stats.str,
   591→                dex = stats.dex,
   592→                int = stats.int,
   593→                level = level,
   594→        })
   595→        local crit = rng:NextNumber() < Balance.critChance(stats.dex)
   596→        if crit then
   597→                atk = math.floor(atk * Balance.CritMult + 0.5)
   598→        end
   599→        -- kurva pelunakan dengan pertahanan monster
   600→        local def = entity.def.def or 0
   601→        local dmg = math.max(1, math.floor(atk * (1 - def / (def + 60)) + 0.5))
   602→
   603→        entity.hp -= dmg
   604→        local bb = entity.bb
   605→        bb.bar.Size = UDim2.fromScale(math.clamp(entity.hp / entity.maxHp, 0, 1), 1)
   606→        bb.label.Text = math.max(0, entity.hp) .. " / " .. entity.maxHp
   607→
   608→        -- krediting boss raid
   609→        if entity.isBoss then
   610→                bossCredits[player] = (bossCredits[player] or 0) + dmg
   611→        end
   612→
   613→        local killed = entity.hp <= 0
   614→        damageTag(entity, dmg, crit, killed)
   615→
   616→        -- recoil visual: angka tersentak sedikit (juice)
   617→        do
   618→                local rootPart = entity.body
   619→                if rootPart and rootPart.Parent and not killed then
   620→                        local dir = (rootPart.Position - root.Position)
   621→                        dir = Vector3.new(dir.X, 0, dir.Z)
   622→                        if dir.Magnitude > 0.1 then
   623→                                dir = dir.Unit
   624→                        else
   625→                                dir = root.CFrame.LookVector * -1
   626→                        end
   627→                        local base = rootPart.Position
   628→                        entity.model:PivotTo(CFrame.new(base + dir * 1.1))
   629→                        task.delay(0.12, function()
   630→                                if entity.alive and entity.model.Parent and entity.body.Parent then
   631→                                        entity.model:PivotTo(CFrame.new(base))
   632→                                end
   633→                        end)
   634→                end
   635→        end
   636→
   637→        if killed then
   638→                CombatService.onKill(player, entity)
   639→                if entity.isBoss then
   640→                        -- bagi hadiah raid SEBELUM boss dibersihkan
   641→                        CombatService.finishBoss()
   642→                end
   643→                monsterDie(entity, player)
   644→        else
   645→                -- monster berbalik mengejar penyerang
   646→                entity.target = player
   647→        end
   648→
   649→        DataService.push(player)
   650→        return { ok = true, dmg = dmg, crit = crit, killed = killed, name = entity.def.name }
   651→end

   653→function CombatService.onKill(player: Player, entity: any)
   654→        local def = entity.def
   655→
   656→        -- XP & Micda (ganda saat PESTA PANEN)
   657→        local xpGain, coinGain = def.xp, rng:NextInteger(def.coin[1], def.coin[2])
   658→        if eventActive then
   659→                xpGain = math.floor(xpGain * Balance.Event.xpMult)
   660→                coinGain = math.floor(coinGain * Balance.Event.coinMult)
   661→        end
   662→        local before = DataService.levelOf(player)
   663→        DataService.addXP(player, xpGain)
   664→        DataService.addCoins(player, coinGain, true)
   665→        if DataService.levelOf(player) > before then
   666→                Remotes.get("Notify"):FireClient(player, "LEVEL NAIK!",
   667→                        "Level " .. DataService.levelOf(player) .. "! Dapat 2 poin stat - bagikan di panel Kar.")
   668→        end
   669→
   670→        -- progres tugas "kalahkan monster"
   671→        QuestService.progress(player, "kill", 1)
   672→
   673→        -- drop material -> bahan craft
   674→        local dropTexts = {}
   675→        for _, drop in ipairs(def.drops) do
   676→                if rng:NextNumber() <= drop.chance then
   677→                        local n = rng:NextInteger(drop.min, drop.max)
   678→                        DataService.giveItem(player, "bag", drop.id, n)
   679→                        table.insert(dropTexts, "+" .. n .. " " .. drop.id)
   680→                end
   681→        end
   682→
   683→        local msg = def.name .. " dikalahkan! +" .. xpGain .. " XP, +" .. coinGain .. " Micda"
   684→        if #dropTexts > 0 then
   685→                msg = msg .. " (" .. table.concat(dropTexts, ", ") .. ")"
   686→        end
   687→        notify(player, "KOMBAT!", msg)
   688→
   689→        -- boss raid: broadcast peristiwa
   690→        if entity.isBoss then
   691→                Remotes.get("BossState"):FireAllClients({ active = false, killed = true, name = def.name })
   692→        end
   693→end

   699→local function stepMonster(entity: any, dt: number)
   700→        if not entity.alive or not entity.body.Parent then
   701→                return
   702→        end
   703→        local def = entity.def
   704→
   705→        -- validasi target
   706→        local targetPlayer = entity.target
   707→        if targetPlayer then
   708→                local valid = targetPlayer.Parent ~= nil and isAliveCharacter(targetPlayer)
   709→                if valid then
   710→                        local root = targetPlayer.Character.PrimaryPart
   711→                        local dist = (root.Position - entity.home).Magnitude
   712→                        if dist > def.aggro * 2.2 then
   713→                                valid = false -- terlalu jauh dari rumahnya, pulang
   714→                        end
   715→                end
   716→                if not valid then
   717→                        entity.target = nil
   718→                end
   719→        end
   720→
   721→        -- cari target baru dalam radius aggro
   722→        if not entity.target then
   723→                for _, player in Players:GetPlayers() do
   724→                        if isAliveCharacter(player) and DataService.has(player) then
   725→                                local root = player.Character.PrimaryPart
   726→                                if (root.Position - entity.body.Position).Magnitude <= def.aggro then
   727→                                        entity.target = player
   728→                                        break
   729→                                end
   730→                        end
   731→                end
   732→        end
   733→
   734→        local bodyPos = entity.body.Position
   735→        local goal: Vector3
   736→        local speed = def.speed
   737→
   738→        if entity.target then
   739→                local root = entity.target.Character.PrimaryPart
   740→                local dist = (root.Position - bodyPos).Magnitude
   741→                if dist > 5 then
   742→                        goal = root.Position
   743→                else
   744→                        goal = bodyPos -- dekat: berhenti & serang
   745→                        local now = os.clock()
   746→                        if now - entity.lastHit >= 1.5 then
   747→                                entity.lastHit = now
   748→                                -- damage ke pemain dengan pertahanan zirah + VIT
   749→                                local pstats = statsOf(entity.target)
   750→                                local defense = Balance.playerDefense(EquipService.armorDef(entity.target), pstats.vit)
   751→                                local taken = Balance.damageTaken(def.atk, defense)
   752→                                local humanoid = entity.target.Character:FindFirstChildOfClass("Humanoid")
   753→                                if humanoid then
   754→                                        humanoid:TakeDamage(taken)
   755→                                end
   756→                        end
   757→                end
   758→        else
   759→                -- berkeliaran di sekitar rumah
   760→                if (bodyPos - entity.wanderTo).Magnitude < 3 then
   761→                        entity.wanderTo = entity.home + Vector3.new(rng:NextNumber(-12, 12), 0, rng:NextNumber(-12, 12))
   762→                        return
   763→                end
   764→                goal = entity.wanderTo
   765→                speed = def.speed * 0.45
   766→        end
   767→
   768→        local dir = (goal - bodyPos) * Vector3.new(1, 0, 1)
   769→        if dir.Magnitude > 0.4 then
   770→                dir = dir.Unit
   771→                local step = math.min(speed * dt, dir.Magnitude * 99)
   772→                local newPos = bodyPos + dir * step
   773→                -- v4: pindahkan SELURUH model (angka, aura, mata) supaya tidak
   774→                -- ada part tertinggal berserakan (penyebab dunia terlihat kotor)
   775→                entity.model:PivotTo(CFrame.lookAt(newPos, newPos + dir))
   776→        end
   777→end
   778→
   779→RunService.Heartbeat:Connect(function(dt)
   780→        for _, entity in monsters do
   781→                pcall(stepMonster, entity, dt)
   782→        end
   783→end)

   789→function CombatService.startBoss()
   790→        if boss then
   791→                return
   792→        end
   793→        local def = Balance.Boss
   794→        local center = zoneCenter("arena")
   795→        local model, body = buildMonsterModel({ id = def.id, name = def.name, body = def.body }, center + Vector3.new(0, 2, 0), true)
   796→        local maxHp = def.hp
   797→        local bb = hpBillboard(body, def.name, maxHp, maxHp, true)
   798→
   799→        boss = {
   800→                def = def,
   801→                zone = "arena",
   802→                model = model,
   803→                body = body,
   804→                bb = bb,
   805→                hp = maxHp,
   806→                maxHp = maxHp,
   807→                alive = true,
   808→                home = center,
   809→                target = nil,
   810→                lastHit = 0,
   811→                wanderTo = center,
   812→                isBoss = true,
   813→                unique = 0,
   814→                spawnedAt = os.clock(),
   815→        }
   816→        monsterSeq += 1
   817→        boss.unique = monsterSeq
   818→        bossCredits = {}
   819→
   820→        local prompt = body:FindFirstChildOfClass("ProximityPrompt")
   821→        if prompt then
   822→                prompt.Triggered:Connect(function(player)
   823→                        if boss and boss.alive then
   824→                                task.spawn(CombatService.playerAttack, player, boss)
   825→                        end
   826→                end)
   827→        end
   828→        monsters["m" .. boss.unique] = boss
   829→
   830→        Remotes.get("BossState"):FireAllClients({
   831→                active = true, name = def.name, hp = maxHp, maxHp = maxHp,
   832→                zone = "Arena Timur",
   833→        })
   834→        for _, player in Players:GetPlayers() do
   835→                notify(player, "BOSS RAID!", def.name .. " muncul di Arena Timur! Serang bersama untuk hadiah besar!")
   836→        end
   837→end

   839→function CombatService.finishBoss()
   840→        if not boss then
   841→                return
   842→        end
   843→        local def = Balance.Boss
   844→        local participants = {}
   845→        local totalDmg = 0
   846→        for player, dmg in bossCredits do
   847→                if player.Parent then
   848→                        table.insert(participants, { player = player, dmg = dmg })
   849→                        totalDmg += dmg
   850→                end
   851→        end
   852→        table.sort(participants, function(a, b)
   853→                return a.dmg > b.dmg
   854→        end)
   855→
   856→        for rank, entry in participants do
   857→                local share = entry.dmg / math.max(1, totalDmg)
   858→                local coin = math.floor(def.participationCoin + share * (rng:NextInteger(def.coin[1], def.coin[2])))
   859→                local xp = math.max(def.participationXp, math.floor(def.xp * share))
   860→                DataService.addCoins(entry.player, coin, true)
   861→                DataService.addXP(entry.player, xp)
   862→                local extra = ""
   863→                if rank == 1 then
   864→                        DataService.addCoins(entry.player, def.topBonusCoin, true)
   865→                        for _, drop in ipairs(def.drops) do
   866→                                local n = rng:NextInteger(drop.min, drop.max)
   867→                                DataService.giveItem(entry.player, "bag", drop.id, n)
   868→                                extra = extra .. " +" .. n .. " " .. drop.id
   869→                        end
   870→                end
   871→                notify(entry.player, "BOSS DIKALAHKAN!", rank == 1 and ("KAU PENYERANG TERAKHIR/TERBESAR! +" .. coin .. " Micda" .. extra)
   872→                        or ("Kontribusi " .. math.floor(share * 100) .. "% - +" .. coin .. " Micda, +" .. xp .. " XP"))
   873→                DataService.push(entry.player)
   874→        end
   875→        bossCredits = {}
   876→end

   878→local function bossLoop()
   879→        while true do
   880→                task.wait(5)
   881→                local now = os.clock()
   882→                if not boss and now >= bossNextSpawn then
   883→                        CombatService.startBoss()
   884→                end
   885→                if boss and boss.alive then
   886→                        -- despawn kalau terlalu lama tak dibunuh
   887→                        if now - boss.spawnedAt > Balance.Boss.despawn then
   888→                                bossCredits = {}
   889→                                monsterDie(boss)
   890→                                boss = nil
   891→                                Remotes.get("BossState"):FireAllClients({ active = false, name = Balance.Boss.name })
   892→                                Remotes.get("Notify"):FireAllClients("Boss Pergi", "Angka Sang Raja kembali ke tanah. Hati-hati, ia akan bangkit lagi!")
   893→                        else
   894→                                Remotes.get("BossState"):FireAllClients({
   895→                                        active = true, name = Balance.Boss.name,
   896→                                        hp = math.max(0, boss.hp), maxHp = boss.maxHp,
   897→                                        zone = "Arena Timur",
   898→                                })
   899→                        end
   900→                end
   901→                if not boss then
   902→                        bossNextSpawn = math.max(bossNextSpawn, now + Balance.Boss.respawn)
   903→                end
   904→        end
   905→end

   911→function CombatService.spawnMonsters()
   912→        for _, zone in ipairs(Balance.Zones) do
   913→                for _, spec in ipairs(zone.monsters) do
   914→                        local def = nil
   915→                        for _, m in ipairs(Balance.Monsters) do

   917→                                        def = m
   918→                                        break
   919→                                end
   920→                        end
   921→                        if def then
   922→                                for _ = 1, spec.count do
   923→                                        spawnMonster(def, zone.id)
   924→                                end
   925→                        end
   926→                end
   927→        end
   928→        print("[AKIOMIDA] Monster dikerahkan di 4 zona.")
   929→end

   961→                local entity = monsters[monsterId] or boss
   962→                if not entity or not entity.alive or ("m" .. entity.unique) ~= monsterId then
   963→                        return { ok = false, msg = "Target tidak ada." }

   974→        bossNextSpawn = os.clock() + 360
   975→        task.spawn(bossLoop)
   976→        task.spawn(eventLoop)
