    27→local function part(props): Part
    28→        local p = Instance.new("Part")
    29→        p.Anchored = true
    30→        p.TopSurface = Enum.SurfaceType.Smooth
    31→        p.BottomSurface = Enum.SurfaceType.Smooth
    32→        for k, v in props do
    33→                (p :: any)[k] = v
    34→        end
    35→        p.Parent = folder
    36→        return p
    37→end

   344→local function buildTerrain()
   345→        local terrain = workspace.Terrain
   346→        terrain.WaterWaveSize = 0.16
   347→        terrain.WaterWaveSpeed = 16
   348→        terrain.WaterReflectance = 0.4
   349→        terrain.WaterTransparency = 0.6
   350→
   351→        -- LAUT DULU (top y=-1), lalu daratan menimpanya di tiap pulau
   352→        terrain:FillBlock(CFrame.new(260, -7, 0), Vector3.new(3400, 12, 3000), Enum.Material.Water)
   353→
   354→        -- daratan tiap pulau wilayah (top y=0)
   355→        for _, r in ipairs(Balance.Regions) do
   356→                local cx, cz = r.center[1], r.center[2]
   357→                local hx, hz = r.half[1], r.half[2]
   358→                terrain:FillBlock(CFrame.new(cx, -5.1, cz), Vector3.new(hx * 2 + 14, 9.5, hz * 2 + 14), Enum.Material.Sand)
   359→                terrain:FillBlock(CFrame.new(cx, -5, cz), Vector3.new(hx * 2, 10, hz * 2), Enum.Material.Grass)
   360→        end
   361→
   362→        -- pulau kecil dekorasi cakrawala (panorama, tanpa fungsi)
   363→        for _, pos in {
   364→                { -320, 120 }, { 380, -720 }, { -280, -820 }, { 1180, -120 }, { 420, 950 }, { -180, 880 },
   365→        } do
   366→                terrain:FillBlock(CFrame.new(pos[1], -5.1, pos[2]), Vector3.new(74, 9.5, 74), Enum.Material.Sand)
   367→                terrain:FillBlock(CFrame.new(pos[1], -5, pos[2]), Vector3.new(60, 10, 60), Enum.Material.Grass)
   368→        end
   369→end
