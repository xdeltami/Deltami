    39→local function fmt(n: number): string
    40→        local s = tostring(math.floor(n))
    41→        local rev = string.reverse(s)
    42→        rev = rev:gsub("(%d%d%d)", "%1.")
    43→        rev = rev:gsub("%.$", "")
    44→        return string.reverse(rev)
    45→end

   554→local function semaiPopBuka(id: string)
   555→        local info = CropData.byId[id]
   556→        if not info then
   557→                return
