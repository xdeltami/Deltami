   121→local function tanamanDemo(parent: Instance, x: number, z: number, cropId: string)
   122→        local b = CropData.Bentuk[cropId]
   123→        if not b then
   124→                return

   127→        local tinggi = math.min(b.t, 2.6)

   133→                local dia = math.min(b.w, 2)
   134→                local d = part(parent, Vector3.new(dia, dia, dia), CFrame.new(x, GT + dia / 2, z), w, Enum.Material.Grass)
   135→                d.Shape = Enum.PartType.Ball
   136→        else
   137→                part(parent, Vector3.new(math.max(b.w * 0.5, 0.5), tinggi, math.max(b.w * 0.5, 0.5)), CFrame.new(x, GT + tinggi / 2, z), w, Enum.Material.Grass)

   323→local function dalamSkip(skips, v: number): boolean
   324→        if not skips then
   325→                return false
   326→        end
   327→        for _, r in ipairs(skips) do
   328→                if v >= r[1] and v <= r[2] then
   329→                        return true
   330→                end
   331→        end
   332→        return false
   333→end

   336→-- dua sisi + jalur sepeda merah dua sisi. Sumbu jalan mengikuti sisi
   337→-- terpanjang dari (ax,az)->(bx,bz).
   338→local function jalanAspal(parent: Instance, ax: number, az: number, bx: number, bz: number, o)
   339→        o = o or {}
   340→        local lebar = o.lebar or 12
   341→        local tebal = o.tebal or 0.2
   342→        local horizontal = math.abs(bx - ax) >= math.abs(bz - az)
   343→        local len = horizontal and (bx - ax) or (bz - az)
   344→        local abslen = math.abs(len)
   345→        local cx, cz = (ax + bx) / 2, (az + bz) / 2
   346→        local m = Instance.new("Model")
   347→        m.Name = "Jalan_" .. (o.nama or "Aspal")
   348→        m.Parent = parent
   349→        local rot = horizontal and 0 or math.rad(90)
   350→        local cf0 = CFrame.new(cx, GT, cz) * CFrame.Angles(0, rot, 0)
   351→        local aspal = part(m, Vector3.new(abslen, tebal, lebar), cf0 * CFrame.new(0, tebal / 2, 0), ASPAL, Enum.Material.Asphalt)
   352→        aspal.Name = "JalanAspal"
   353→        local nDash = math.floor((abslen - 6) / 7)
   354→        for i = 0, nDash - 1 do
   355→                local along = -abslen / 2 + 6 + i * 7
   356→                if math.abs(along) < abslen / 2 - 2 then
   357→                        local dash = part(m, Vector3.new(2.6, 0.06, 0.45), cf0 * CFrame.new(along, tebal + 0.07, 0), MARKA, Enum.Material.SmoothPlastic)
   358→                        dash.Name = "MarkaPutih"
   359→                        dash.CanCollide = false
   360→                end
   361→        end
   362→        for _, s in ipairs({ -1, 1 }) do
   363→                local trotoar = part(m, Vector3.new(abslen, 0.35, 2.4), cf0 * CFrame.new(0, 0.175, s * (lebar / 2 + 0.8)), TROTOAR, Enum.Material.Concrete)
   364→                trotoar.Name = "Trotoar"
   365→                local jalur = part(m, Vector3.new(abslen, 0.08, 2), cf0 * CFrame.new(0, tebal + 0.05, s * (lebar / 2 - 1.5)), SEPEDA, Enum.Material.SmoothPlastic)
   366→                jalur.Name = "JalurSepeda"
   367→                jalur.CanCollide = false
   368→        end
   369→        return m, horizontal, cx, cz, abslen
   370→end

   372→-- Lampu jalan + pohon pinggir jalan, bolak-balik sisi, dengan daftar
   373→-- rentang koordinat yang dilewati (persimpangan/zebra/gerbang).
   374→local function hiasJalan(parent: Instance, horizontal: boolean, cx: number, cz: number, abslen: number, lebar: number, skips)
   375→        local s = -abslen / 2 + 14
   376→        local k = 0
   377→        while s <= abslen / 2 - 14 do
   378→                local pusatSum = horizontal and cx or cz
   379→                if not dalamSkip(skips, pusatSum + s) then
   380→                        local side = (k % 2 == 0) and -1 or 1
   381→                        local lx, lz
   382→                        if horizontal then
   383→                                lx, lz = cx + s, cz + side * (lebar / 2 + 1.2)
   384→                        else
   385→                                lx, lz = cx + side * (lebar / 2 + 1.2), cz + s
   386→                        end
   387→                        lampuTaman(parent, lx, lz)
   388→                        local ta = s + 15
   389→                        if not dalamSkip(skips, pusatSum + ta) then
   390→                                local tx, tz
   391→                                if horizontal then
   392→                                        tx, tz = cx + ta, cz - side * (lebar / 2 + 2.8)
   393→                                else
   394→                                        tx, tz = cx - side * (lebar / 2 + 2.8), cz + ta
   395→                                end
   396→                                pohonAlam(parent, tx, tz, 0.72)
   397→                        end
   398→                end
   399→                s += 30
   400→                k += 1
   401→        end
   402→end
