    25→local function fmt(n: number): string
    26→        local s = tostring(math.floor(n))
    27→        local rev = string.reverse(s)
    28→        rev = rev:gsub("(%d%d%d)", "%1.")
    29→        rev = rev:gsub("%.$", "")
    30→        return string.reverse(rev)
