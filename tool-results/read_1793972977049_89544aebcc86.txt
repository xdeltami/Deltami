    31→local function fmt(n: number): string
    32→        local s = tostring(math.floor(n))
    33→        local rev = string.reverse(s)
    34→        rev = rev:gsub("(%d%d%d)", "%1.")
    35→        rev = rev:gsub("%.$", "")
    36→        return string.reverse(rev)
    37→end
