    24→local function part(props): Part
    25→        local p = Instance.new("Part")
    26→        p.Anchored = true
    27→        p.TopSurface = Enum.SurfaceType.Smooth
    28→        p.BottomSurface = Enum.SurfaceType.Smooth
    29→        for k, v in props do
    30→                (p :: any)[k] = v
    31→        end
    32→        p.Parent = folder
    33→        return p
    34→end
