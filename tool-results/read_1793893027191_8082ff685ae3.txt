   327→function Balance.koperasiName(rng: Random?): string
   328→        local r = rng or Random.new()
   329→        local code = table.create(8)
   330→        for i = 1, 8 do
   331→                local n = r:NextInteger(1, #KOPERASI_CHARS)

   333→        end
   334→        return Balance.KOPERASI_PREFIX .. table.concat(code)
   335→end

   337→--- Bersihkan nama lahan dari papan nama: 3..20 karakter aman (murni).
   338→function Balance.sanitizeLandName(text: any): string?
   339→        if type(text) ~= "string" then
   340→                return nil
   341→        end
   342→        local cleaned = string.gsub(text, "%s+", " ")
   343→        cleaned = string.gsub(cleaned, "^%s+", ""):gsub("%s+$", "")
   344→        if #cleaned < 3 or #cleaned > 20 then

   362→--- Titik muncul di region tujuan: 13 stud ke dalam dari gerbang yang
   363→--- menghadap region asal (murni - diuji offline). "home" cocok dengan
   364→--- "rumahK" mana pun.
   365→function Balance.gateArrival(regionId: string, fromRegionId: string): (number?, number?)
   366→        local region = Balance.regionById(regionId)
   367→        if not region then
   368→                return nil, nil
   369→        end
   370→        for _, g in ipairs(region.gates) do
   371→                local match = g.dest == fromRegionId
   372→                        or (g.dest == "home" and string.sub(fromRegionId, 1, 5) == "rumah")
   373→                if match then
   374→                        local dx = region.center[1] - g.pos[1]
   375→                        local dz = region.center[2] - g.pos[2]
   376→                        local len = math.sqrt(dx * dx + dz * dz)
   377→                        if len < 1 then
   378→                                return g.pos[1], g.pos[2]
   379→                        end
   380→                        return g.pos[1] + dx / len * 13, g.pos[2] + dz / len * 13
   381→                end
   382→        end
   383→        return nil, nil
   384→end

   386→--- Dua kotak (center1, half1, center2, half2) bertumpuk dengan margin (murni).
   387→function Balance.rectsOverlap(c1: { number }, h1: { number }, c2: { number }, h2: { number }, margin: number): boolean
   388→        return math.abs(c1[1] - c2[1]) < h1[1] + h2[1] + margin
   389→                and math.abs(c1[2] - c2[2]) < h1[2] + h2[2] + margin
   390→end

   392→--- Validasi arsitektur wilayah (murni - dipanggil tes offline):
   393→--- gerbang menunjuk wilayah yang ada, posisi gerbang di dalam pulau,
   394→--- zona monster di dalam pulau combat & TIDAK menyentuh pulau peradaban.
   395→function Balance.validateRegions(): { string }
   396→        local errs = {}
   397→        local ids = {}
   398→        for _, r in ipairs(Balance.Regions) do
   399→                if ids[r.id] then
   400→                        table.insert(errs, "region duplikat: " .. r.id)
   401→                end
   402→                ids[r.id] = true
   403→        end
   404→        for _, r in ipairs(Balance.Regions) do
   405→                for _, g in ipairs(r.gates) do
   406→                        if g.dest ~= "home" and not ids[g.dest] then
   407→                                table.insert(errs, "gerbang " .. r.id .. " menunjuk wilayah tak dikenal: " .. tostring(g.dest))
   408→                        end
   409→                        if math.abs(g.pos[1] - r.center[1]) > r.half[1] - 4
   410→                                or math.abs(g.pos[2] - r.center[2]) > r.half[2] - 4 then
   411→                                table.insert(errs, "gerbang " .. r.id .. " keluar batas pulau")
   412→                        end
   413→                end
   414→        end
   415→        for _, zone in ipairs(Balance.Zones) do
   416→                local island
   417→                for _, r in ipairs(Balance.Regions) do
   418→                        if r.kind == "combat" and Balance.rectsOverlap(
   419→                                { zone.center[1], zone.center[3] }, { zone.radius, zone.radius },
   420→                                r.center, r.half, 0) then
   421→                                island = r
   422→                                break
   423→                        end
   424→                end
   425→                if not island then
   426→                        table.insert(errs, "zona monster '" .. zone.id .. "' tidak berada di pulau combat mana pun")
   427→                else
   428→                        if math.abs(zone.center[1] - island.center[1]) + zone.radius > island.half[1] - 10
   429→                                or math.abs(zone.center[3] - island.center[2]) + zone.radius > island.half[2] - 10 then
   430→                                table.insert(errs, "zona monster '" .. zone.id .. "' menjulur keluar pulau " .. island.id)
   431→                        end
   432→                end
   433→                for _, r in ipairs(Balance.Regions) do
   434→                        if r.kind ~= "combat" and Balance.rectsOverlap(
   435→                                { zone.center[1], zone.center[3] }, { zone.radius, zone.radius },
   436→                                r.center, r.half, 12) then
   437→                                table.insert(errs, "zona monster '" .. zone.id .. "' menembus wilayah " .. r.kind .. " '" .. r.id .. "'")
   438→                        end
   439→                end
   440→        end
   441→        for _, m in ipairs(Balance.Monsters) do
   442→                local ok = false
   443→                for _, z in ipairs(Balance.Zones) do
   444→                        if z.id == m.zone then
   445→                                ok = true
   446→                                break
   447→                        end
   448→                end
   449→                if not ok then
   450→                        table.insert(errs, "monster '" .. m.id .. "' zona tidak dikenal: " .. tostring(m.zone))
   451→                end
   452→        end
   453→        return errs
   454→end

   477→--- Rotasi harian deterministik (sama di server & client): pilih
   478→--- QuestsPerDay tugas dari kolam berdasarkan hash hari ini.
   479→function Balance.todayQuestIds(): { string }
   480→        local day = math.floor(os.time() / 86400)
   481→        local pool = {}
   482→        for _, q in ipairs(Balance.Quests) do
   483→                table.insert(pool, q.id)
   484→        end
   485→        local function hashString(s: string): number
   486→                local h = 0
   487→                for i = 1, #s do
   488→                        h = (h * 31 + string.byte(s, i)) % 1000000007
   489→                end
   490→                return h
   491→        end
   492→        local picked: { string } = {}
   493→        local start = day % #pool
   494→        for i = 1, Balance.QuestsPerDay do
   495→                local idx = ((start + (hashString("akm" .. day) + i * 7) % #pool)) % #pool
   496→                local id = pool[idx + 1]
   497→                local dup = false
   498→                for _, p in picked do
   499→                        if p == id then
   500→                                dup = true
   501→                                break
   502→                        end
   503→                end
   504→                if not dup then
   505→                        table.insert(picked, id)
   506→                end
   507→        end
   508→        -- fallback: kalau kurang (tabrakan duplikat), isi berurutan
   509→        for _, id in pool do
   510→                if #picked >= Balance.QuestsPerDay then
   511→                        break
   512→                end
   513→                local dup = false
   514→                for _, p in picked do
   515→                        if p == id then
   516→                                dup = true
   517→                                break
   518→                        end
   519→                end
   520→                if not dup then
   521→                        table.insert(picked, id)
   522→                end
   523→        end
   524→        return picked
   525→end

   604→--- Kurva kekuatan: simulasi waktu-bunuh monster oleh pemain level L.
   605→--- Dipakai untuk cek keseimbangan (TTK wajar: tidak instan, tidak lama).
   606→function Balance.simulateTtk(level: number, monsterId: string, statSpread: string?): (number, number)
   607→        local spread = statSpread or "balanced"
   608→        local str, dex, vit, agi, int = 0, 0, 0, 0, 0
   609→        local points = (level - 1) * 2
   610→        if spread == "balanced" then
   611→                str = math.floor(points * 0.35)
   612→                dex = math.floor(points * 0.25)
   613→                vit = math.floor(points * 0.25)
   614→                agi = points - str - dex - vit
   615→        elseif spread == "attacker" then
   616→                str = math.floor(points * 0.7)
   617→                dex = math.floor(points * 0.3)
   618→        else
   619→                vit = math.floor(points * 0.7)
   620→                agi = points - vit
   621→        end
   622→        local atk = Balance.playerAttack({ weaponAtk = 14, str = str, dex = dex, int = int, level = level })
   623→        local crit = Balance.critChance(dex)
   624→        local dps = atk * (1 + crit * (Balance.CritMult - 1)) / Balance.attackCooldown(agi)
   625→        local mon = nil
   626→        for _, m in ipairs(Balance.Monsters) do
   627→                if m.id == monsterId then
   628→                        mon = m
   629→                        break
   630→                end
   631→        end
   632→        if not mon then
   633→                return -1, -1
   634→        end
   635→        local ttk = mon.hp / math.max(1, dps)
   636→        -- berapa damage pemain terima sebelum monster mati
   637→        local def = Balance.playerDefense(8, vit)
   638→        local hits = math.ceil(ttk / 1.5) -- monster menyerang tiap ~1.5s
   639→        return ttk, Balance.damageTaken(mon.atk, def) * hits
   640→end
