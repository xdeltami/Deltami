    29→local function fmt(n: number): string
    30→        local s = tostring(math.floor(n))
    31→        local rev = string.reverse(s)
    32→        rev = rev:gsub("(%d%d%d)", "%1.")
    33→        rev = rev:gsub("%.$", "")
    34→        return string.reverse(rev)
    35→end

   209→local function ikonSlot(cell: Frame, s: any?)
   210→        local lama = cell:FindFirstChild("Ikon")
   211→        if lama then
   212→                lama:Destroy()
   213→        end

   218→        if s.kind == "blok" then
   219→                mk("Frame", { Size = UDim2.fromScale(0.62, 0.62), Position = UDim2.fromScale(0.19, 0.19), BackgroundColor3 = warna(UIStyle.AbuGelap), BorderSizePixel = 0 }, wadah)

   223→        local b = info and CropData.Bentuk[info.id]
   224→        local buahC = warna3b(b and b.siap)
   225→        local aksenC = warna3b(b and b.aksen)
   226→        local bagian = DropBentuk.ikonBagian(b and b.jenis or "daun") or {}
   227→        for _, g in ipairs(bagian) do
   228→                local f = mk("Frame", { Position = UDim2.fromScale(g.dx, g.dy), Size = UDim2.fromScale(g.dw, g.dh), BackgroundColor3 = g.warna == "buah" and buahC or aksenC, BorderSizePixel = 0, ZIndex = 2 }, wadah)
   229→                if g.bentuk == "lingkaran" then
   230→                        mk("UICorner", { CornerRadius = UDim.new(1, 0) }, f)
   231→                elseif g.bentuk == "topi" then
   232→                        mk("UICorner", { CornerRadius = UDim.new(0.5, 0) }, f)

   235→                end
