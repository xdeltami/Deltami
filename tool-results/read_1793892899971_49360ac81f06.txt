   169→local function updateTilePrompt(tile)
   170→        local prompt = tile.prompt
   171→        if not prompt then
   172→                return
   173→        end
   174→        local plot = plots[tile.plotIndex]
   175→        local owner = plot and plot.owner
   176→
   177→        if not owner then
   178→                prompt.Enabled = false
   179→                return
   180→        end
   181→
   182→        if tile.state == "empty" then
   183→                prompt.ActionText = "Tanam Bibit"
   184→                prompt.ObjectText = "Petak Ladang"
   185→                prompt.HoldDuration = 0
   186→                prompt.Enabled = true
   187→        elseif tile.state == "growing" then
   188→                local data = DataService.get(owner)
   189→                local canWater = data.Tools.can and not tile.watered
   190→                if canWater then
   191→                        prompt.ActionText = "Siram"
   192→                        prompt.ObjectText = tile.seed .. " tumbuh"
   193→                        prompt.HoldDuration = 0.4
   194→                        prompt.Enabled = true
   195→                else
   196→                        prompt.Enabled = false
   197→                end
   198→        elseif tile.state == "ready" then
   199→                prompt.ActionText = "Panen"
   200→                prompt.ObjectText = cropById[tile.seed] and cropById[tile.seed].name or "Hasil"
   201→                prompt.HoldDuration = 0
   202→                prompt.Enabled = true
   203→        end
   204→end
   205→
   206→local function onTileTriggered(player: Player, tile)
   207→        local plot = plots[tile.plotIndex]
   208→        if not plot or plot.owner ~= player or not DataService.has(player) then
   209→                return
   210→        end
   211→
   212→        if tile.state == "empty" then
   213→                Remotes.get("OpenPlant"):FireClient(player, tile.plotIndex, tile.tileIndex)
   214→        elseif tile.state == "growing" then
   215→                local data = DataService.get(player)
   216→                if not data.Tools.can then
   217→                        Remotes.get("Notify"):FireClient(player, "Butuh Alat", "Kamu butuh Kaleng Siram - beli di Toko Alat.")
   218→                        return
   219→                end
   220→                if tile.watered then
   221→                        return
   222→                end
   223→                tile.watered = true
   224→                tile.plantedAt -= tile.growTime * Config.Economy.WaterBoost
   225→                Remotes.get("Notify"):FireClient(player, "Disiram!", "Tanamanmu tumbuh 25% lebih cepat.")
   226→                DataService.addXP(player, 2)
   227→                DataService.push(player)
   228→        elseif tile.state == "ready" then
   229→                local crop = cropById[tile.seed]
   230→                if not crop then
   231→                        return
   232→                end
   233→                -- v20: percikan panen (sekali pakai, ringan untuk mobile)
   234→                Util.burst(tile.part, Color3.fromRGB(crop.color[1], crop.color[2], crop.color[3]), 10, 6)
   235→                DataService.giveItem(player, "crop", crop.id, 1)
   236→                DataService.addXP(player, 4)
   237→                QuestService.progress(player, "harvest", 1)
   238→                tile.state = "empty"
   239→                tile.seed = nil
   240→                tile.plantedAt = 0
   241→                tile.growTime = 0
   242→                tile.watered = false
   243→                tile.bucket = -1
   244→                destroyPlant(tile)
   245→                updateTilePrompt(tile)
   246→                DataService.push(player)
   247→        end
   248→end

   258→local function updateSign(plot)
   259→        local signText = plot.signText
   260→        if not signText then
   261→                return
   262→        end
   263→        if plot.owner then
   264→                signText.Text = "LADANG " .. plot.index .. " - MILIK " .. string.upper(plot.owner.DisplayName)
   265→                plot.claimPrompt.Enabled = false
   266→        else
   267→                local price = priceOfPlot(plot.index)
   268→                signText.Text = price <= 0
   269→                        and "LADANG " .. plot.index .. " - AMBIL ALIH (GRATIS)"
   270→                        or "LADANG " .. plot.index .. " - BELI " .. price .. " MICDA"
   271→                plot.claimPrompt.Enabled = true
   272→        end
   273→end

   275→local function onClaimTriggered(player: Player, plot)
   276→        if plot.owner then
   277→                if plot.owner == player then
   278→                        Remotes.get("Notify"):FireClient(player, "Ladangmu", "Ini sudah ladang milikmu. Tanami bibit di petak!")
   279→                end
   280→                return
   281→        end
   282→        -- v4 EKONOMI SYARIAH: ladang 1 GRATIS; ladang lain DIBELI dengan Micda
   283→        -- (bayar penuh di muka - tanpa cicilan, tanpa bunga, tanpa riba)
   284→        local price = priceOfPlot(plot.index)
   285→        if price > 0 then
   286→                local data = DataService.get(player)
   287→                if data.Coins < price then
   288→                        Remotes.get("Notify"):FireClient(player, "Micda Kurang",
   289→                                "Ladang " .. plot.index .. " dijual " .. price .. " Micda (bayar penuh, tanpa bunga). Micdamu: " .. data.Coins .. ".")
   290→                        return
   291→                end
   292→                DataService.addCoins(player, -price, false)
   293→        end
   294→        plot.owner = player
   295→        updateSign(plot)
   296→        for _, tile in plot.tiles do
   297→                updateTilePrompt(tile)
   298→        end
   299→        local msg = price > 0
   300→                and "Ladang " .. plot.index .. " kini milikmu (dibeli " .. price .. " Micda, lunas)! Beli bibit di Toko Bibit, lalu dekati petak tanah untuk menanam."
   301→                or "Ladang " .. plot.index .. " kini milikmu (GRATIS)! Beli bibit di Toko Bibit (Pasar Timur), lalu dekati petak tanah untuk menanam."
   302→        Remotes.get("Notify"):FireClient(player, "Ladang Baru!", msg)
   303→        DataService.push(player)
   304→end

   499→local function onPlantSeed(player: Player, plotIndex: any, tileIndex: any, seedId: any)
   500→        if not DataService.has(player) then
   501→                return { ok = false, msg = "Data belum siap." }
   502→        end
   503→        if type(plotIndex) ~= "number" or type(tileIndex) ~= "number" or type(seedId) ~= "string" then
   504→                return { ok = false, msg = "Permintaan tidak valid." }
   505→        end
   506→        local plot = plots[math.floor(plotIndex)]
   507→        local tile = plot and plot.tiles[math.floor(tileIndex)]
   508→        if not plot or not tile or plot.owner ~= player then
   509→                return { ok = false, msg = "Petak tidak tersedia." }
   510→        end
   511→        if tile.state ~= "empty" then
   512→                return { ok = false, msg = "Petak sudah ditanami." }
   513→        end
   514→        local crop = cropById[seedId]
   515→        if not crop then
   516→                return { ok = false, msg = "Bibit tidak dikenal." }
   517→        end
   518→        if DataService.levelOf(player) < crop.level then
   519→                return { ok = false, msg = "Butuh level " .. crop.level .. " untuk bibit ini." }
   520→        end
   521→        if not DataService.takeItem(player, "seed", seedId, 1) then
   522→                return { ok = false, msg = "Bibit habis. Beli lagi di Toko Bibit!" }
   523→        end
   524→
   525→        tile.state = "growing"
   526→        tile.seed = seedId
   527→        tile.plantedAt = time()
   528→        tile.growTime = crop.grow
   529→        tile.watered = false
   530→        tile.bucket = 0
   531→        buildPlantVisual(tile, 0)
   532→        updateTilePrompt(tile)
   533→        DataService.addXP(player, 2)
   534→        DataService.push(player)
   535→        return { ok = true, msg = crop.name .. " ditanam! Kembali setelah ±" .. crop.grow .. " detik." }
   536→end

   542→function FarmService.startLoop()
   543→        task.spawn(function()
   544→                while true do
   545→                        task.wait(2)
   546→                        for _, plot in plots do
   547→                                if not plot.owner then
   548→                                        continue
   549→                                end
   550→                                for _, tile in plot.tiles do
   551→                                        if tile.state ~= "growing" then
   552→                                                continue
   553→                                        end
   554→                                        local progress = (time() - tile.plantedAt) / math.max(1, tile.growTime)
   555→                                        if progress >= 1 then
   556→                                                tile.state = "ready"
   557→                                                buildPlantVisual(tile, 3)
   558→                                                updateTilePrompt(tile)
   559→                                        else
   560→                                                local bucket = math.floor(progress * 3) -- 0..2
   561→                                                if bucket ~= tile.bucket then
   562→                                                        tile.bucket = bucket
   563→                                                        buildPlantVisual(tile, bucket)
   564→                                                end
   565→                                                local prompt = tile.prompt
   566→                                                if prompt and not prompt.Enabled then
   567→                                                        updateTilePrompt(tile) -- misal pemain baru saja beli kaleng siram
   568→                                                end
   569→                                        end
   570→                                end
   571→                        end
   572→                end
   573→        end)
   574→end

   577→-- STATE & PEMBERSIHAN
   578→--------------------------------------------------------------------
   579→
   580→function FarmService.getPlotState(player: Player)
   581→        for _, plot in plots do
   582→                if plot.owner == player then
   583→                        local tilesOut = {}
   584→                        for i, tile in plot.tiles do
   585→                                local entry: any = { state = tile.state }
   586→                                if tile.state ~= "empty" then
   587→                                        entry.seed = tile.seed
   588→                                        entry.watered = tile.watered
   589→                                        if tile.state == "growing" then
   590→                                                entry.remain = math.max(0, math.ceil(tile.growTime - (time() - tile.plantedAt)))
   591→                                        end
   592→                                end
   593→                                tilesOut[i] = entry
   594→                        end
   595→                        return { claimed = true, index = plot.index, tiles = tilesOut }
   596→                end
   597→        end
   598→        return { claimed = false, tiles = {} }
   599→end

   601→--- Pemain keluar: hasil siap panen otomatis masuk tas,
   602→--- tanaman yang masih tumbuh hangus (ladang dibebaskan).
   603→function FarmService.releasePlot(player: Player)
   604→        for _, plot in plots do
   605→                if plot.owner == player then
   606→                        local harvested = 0
   607→                        for _, tile in plot.tiles do
   608→                                if tile.state == "ready" and tile.seed then
   609→                                        DataService.giveItem(player, "crop", tile.seed, 1)
   610→                                        harvested += 1
   611→                                end
   612→                                tile.state = "empty"
   613→                                tile.seed = nil
   614→                                tile.plantedAt = 0
   615→                                tile.growTime = 0
   616→                                tile.watered = false
   617→                                tile.bucket = -1
   618→                                destroyPlant(tile)
   619→                                updateTilePrompt(tile)
   620→                        end
   621→                        if harvested > 0 then
   622→                                Remotes.get("Notify"):FireClient(player, "Panen Otomatis",
   623→                                        harvested .. " hasil siap panen otomatis masuk tas sebelum kamu pergi.")
   624→                        end
   625→                        plot.owner = nil
   626→                        updateSign(plot)
   627→                end
   628→        end
   629→end
