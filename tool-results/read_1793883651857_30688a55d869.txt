   169→local function updateTilePrompt(tile)
   170→        local prompt = tile.prompt
   171→        if not prompt then
   172→                return
   173→        end
   174→        local plot = plots[tile.plotIndex]
   175→        local owner = plot and plot.owner
   176→
   177→        if not owner then
   178→                prompt.Enabled = false
   179→                return
   180→        end
   181→
   182→        if tile.state == "empty" then
   183→                prompt.ActionText = "Tanam Bibit"
   184→                prompt.ObjectText = "Petak Ladang"
   185→                prompt.HoldDuration = 0
   186→                prompt.Enabled = true
   187→        elseif tile.state == "growing" then
   188→                local data = DataService.get(owner)
   189→                local canWater = data.Tools.can and not tile.watered
   190→                if canWater then
   191→                        prompt.ActionText = "Siram"
   192→                        prompt.ObjectText = tile.seed .. " tumbuh"
   193→                        prompt.HoldDuration = 0.4
   194→                        prompt.Enabled = true
   195→                else
   196→                        prompt.Enabled = false
   197→                end
   198→        elseif tile.state == "ready" then
   199→                prompt.ActionText = "Panen"
   200→                prompt.ObjectText = cropById[tile.seed] and cropById[tile.seed].name or "Hasil"
   201→                prompt.HoldDuration = 0
   202→                prompt.Enabled = true
   203→        end
   204→end
   205→
   206→local function onTileTriggered(player: Player, tile)
   207→        local plot = plots[tile.plotIndex]
   208→        if not plot or plot.owner ~= player or not DataService.has(player) then
   209→                return
   210→        end
   211→
   212→        if tile.state == "empty" then
   213→                Remotes.get("OpenPlant"):FireClient(player, tile.plotIndex, tile.tileIndex)
   214→        elseif tile.state == "growing" then
   215→                local data = DataService.get(player)
   216→                if not data.Tools.can then
   217→                        Remotes.get("Notify"):FireClient(player, "Butuh Alat", "Kamu butuh Kaleng Siram - beli di Toko Alat.")
   218→                        return
   219→                end
   220→                if tile.watered then
   221→                        return
   222→                end
   223→                tile.watered = true
   224→                tile.plantedAt -= tile.growTime * Config.Economy.WaterBoost
   225→                Remotes.get("Notify"):FireClient(player, "Disiram!", "Tanamanmu tumbuh 25% lebih cepat.")
   226→                DataService.addXP(player, 2)
   227→                DataService.push(player)
   228→        elseif tile.state == "ready" then
   229→                local crop = cropById[tile.seed]
   230→                if not crop then
   231→                        return
   232→                end
   233→                -- v20: percikan panen (sekali pakai, ringan untuk mobile)
   234→                Util.burst(tile.part, Color3.fromRGB(crop.color[1], crop.color[2], crop.color[3]), 10, 6)
   235→                DataService.giveItem(player, "crop", crop.id, 1)
   236→                DataService.addXP(player, 4)
   237→                QuestService.progress(player, "harvest", 1)
   238→                tile.state = "empty"
   239→                tile.seed = nil
   240→                tile.plantedAt = 0
   241→                tile.growTime = 0
   242→                tile.watered = false
   243→                tile.bucket = -1
   244→                destroyPlant(tile)
   245→                updateTilePrompt(tile)
   246→                DataService.push(player)
   247→        end
   248→end

   258→local function updateSign(plot)
   259→        local signText = plot.signText
   260→        if not signText then
   261→                return

   263→        if plot.owner then
   264→                signText.Text = "LADANG " .. plot.index .. " - MILIK " .. string.upper(plot.owner.DisplayName)
   265→                plot.claimPrompt.Enabled = false
   266→        else
   267→                local price = priceOfPlot(plot.index)
   268→                signText.Text = price <= 0
   269→                        and "LADANG " .. plot.index .. " - AMBIL ALIH (GRATIS)"
   270→                        or "LADANG " .. plot.index .. " - BELI " .. price .. " MICDA"
   271→                plot.claimPrompt.Enabled = true

   275→local function onClaimTriggered(player: Player, plot)
   276→        if plot.owner then
   277→                if plot.owner == player then
   278→                        Remotes.get("Notify"):FireClient(player, "Ladangmu", "Ini sudah ladang milikmu. Tanami bibit di petak!")
   279→                end
   280→                return
   281→        end
   282→        -- v4 EKONOMI SYARIAH: ladang 1 GRATIS; ladang lain DIBELI dengan Micda
   283→        -- (bayar penuh di muka - tanpa cicilan, tanpa bunga, tanpa riba)
   284→        local price = priceOfPlot(plot.index)
   285→        if price > 0 then
   286→                local data = DataService.get(player)
   287→                if data.Coins < price then
   288→                        Remotes.get("Notify"):FireClient(player, "Micda Kurang",
   289→                                "Ladang " .. plot.index .. " dijual " .. price .. " Micda (bayar penuh, tanpa bunga). Micdamu: " .. data.Coins .. ".")
   290→                        return
   291→                end
   292→                DataService.addCoins(player, -price, false)
   293→        end
   294→        plot.owner = player
   295→        updateSign(plot)
   296→        for _, tile in plot.tiles do
   297→                updateTilePrompt(tile)
   298→        end
   299→        local msg = price > 0
   300→                and "Ladang " .. plot.index .. " kini milikmu (dibeli " .. price .. " Micda, lunas)! Beli bibit di Toko Bibit, lalu dekati petak tanah untuk menanam."
   301→                or "Ladang " .. plot.index .. " kini milikmu (GRATIS)! Beli bibit di Toko Bibit (Pasar Timur), lalu dekati petak tanah untuk menanam."
   302→        Remotes.get("Notify"):FireClient(player, "Ladang Baru!", msg)
   303→        DataService.push(player)
   304→end

   484→local function onPlantSeed(player: Player, plotIndex: any, tileIndex: any, seedId: any)
   485→        if not DataService.has(player) then
   486→                return { ok = false, msg = "Data belum siap." }
   487→        end
   488→        if type(plotIndex) ~= "number" or type(tileIndex) ~= "number" or type(seedId) ~= "string" then
   489→                return { ok = false, msg = "Permintaan tidak valid." }
   490→        end
   491→        local plot = plots[math.floor(plotIndex)]
   492→        local tile = plot and plot.tiles[math.floor(tileIndex)]
   493→        if not plot or not tile or plot.owner ~= player then
   494→                return { ok = false, msg = "Petak tidak tersedia." }
   495→        end
   496→        if tile.state ~= "empty" then
   497→                return { ok = false, msg = "Petak sudah ditanami." }
   498→        end
   499→        local crop = cropById[seedId]
   500→        if not crop then
   501→                return { ok = false, msg = "Bibit tidak dikenal." }
   502→        end
   503→        if DataService.levelOf(player) < crop.level then
   504→                return { ok = false, msg = "Butuh level " .. crop.level .. " untuk bibit ini." }
   505→        end
   506→        if not DataService.takeItem(player, "seed", seedId, 1) then
   507→                return { ok = false, msg = "Bibit habis. Beli lagi di Toko Bibit!" }
   508→        end
   509→
   510→        tile.state = "growing"
   511→        tile.seed = seedId
   512→        tile.plantedAt = time()
   513→        tile.growTime = crop.grow
   514→        tile.watered = false
   515→        tile.bucket = 0
   516→        buildPlantVisual(tile, 0)
   517→        updateTilePrompt(tile)
   518→        DataService.addXP(player, 2)
   519→        DataService.push(player)
   520→        return { ok = true, msg = crop.name .. " ditanam! Kembali setelah ±" .. crop.grow .. " detik." }
   521→end

   524→-- LOOP PERTUMBUHAN
   525→--------------------------------------------------------------------
   526→
   527→function FarmService.startLoop()
   528→        task.spawn(function()
   529→                while true do
   530→                        task.wait(2)
   531→                        for _, plot in plots do
   532→                                if not plot.owner then
   533→                                        continue
   534→                                end
   535→                                for _, tile in plot.tiles do
   536→                                        if tile.state ~= "growing" then
   537→                                                continue
   538→                                        end
   539→                                        local progress = (time() - tile.plantedAt) / math.max(1, tile.growTime)
   540→                                        if progress >= 1 then
   541→                                                tile.state = "ready"
   542→                                                buildPlantVisual(tile, 3)
   543→                                                updateTilePrompt(tile)
   544→                                        else
   545→                                                local bucket = math.floor(progress * 3) -- 0..2
   546→                                                if bucket ~= tile.bucket then
   547→                                                        tile.bucket = bucket
   548→                                                        buildPlantVisual(tile, bucket)
   549→                                                end
   550→                                                local prompt = tile.prompt
   551→                                                if prompt and not prompt.Enabled then
   552→                                                        updateTilePrompt(tile) -- misal pemain baru saja beli kaleng siram
   553→                                                end
   554→                                        end
   555→                                end
   556→                        end
   557→                end
   558→        end)
   559→end

   565→function FarmService.getPlotState(player: Player)
   566→        for _, plot in plots do
   567→                if plot.owner == player then
   568→                        local tilesOut = {}
   569→                        for i, tile in plot.tiles do
   570→                                local entry: any = { state = tile.state }
   571→                                if tile.state ~= "empty" then
   572→                                        entry.seed = tile.seed
   573→                                        entry.watered = tile.watered
   574→                                        if tile.state == "growing" then
   575→                                                entry.remain = math.max(0, math.ceil(tile.growTime - (time() - tile.plantedAt)))
   576→                                        end
   577→                                end
   578→                                tilesOut[i] = entry
   579→                        end
   580→                        return { claimed = true, index = plot.index, tiles = tilesOut }
   581→                end
   582→        end
   583→        return { claimed = false, tiles = {} }
   584→end

   586→--- Pemain keluar: hasil siap panen otomatis masuk tas,
   587→--- tanaman yang masih tumbuh hangus (ladang dibebaskan).
   588→function FarmService.releasePlot(player: Player)
   589→        for _, plot in plots do
   590→                if plot.owner == player then
   591→                        local harvested = 0
   592→                        for _, tile in plot.tiles do
   593→                                if tile.state == "ready" and tile.seed then
   594→                                        DataService.giveItem(player, "crop", tile.seed, 1)
   595→                                        harvested += 1
   596→                                end
   597→                                tile.state = "empty"
   598→                                tile.seed = nil
   599→                                tile.plantedAt = 0
   600→                                tile.growTime = 0
   601→                                tile.watered = false
   602→                                tile.bucket = -1
   603→                                destroyPlant(tile)
   604→                                updateTilePrompt(tile)
   605→                        end
   606→                        if harvested > 0 then
   607→                                Remotes.get("Notify"):FireClient(player, "Panen Otomatis",
   608→                                        harvested .. " hasil siap panen otomatis masuk tas sebelum kamu pergi.")
   609→                        end
   610→                        plot.owner = nil
   611→                        updateSign(plot)
   612→                end
   613→        end
   614→end
