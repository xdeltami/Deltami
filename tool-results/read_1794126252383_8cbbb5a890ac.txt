    44→function BuildingKit.gableRoof(parent: Instance, w: number, d: number, oh: number, y0: number, color: Color3)
    45→        local slope = math.rad(30)
    46→        local hd = d / 2 + oh
    47→        local rise = hd * math.tan(slope)
    48→        local slabLen = hd / math.cos(slope)
    49→        BuildingKit.part(parent, Vector3.new(w + 2 * oh, 1, slabLen), CFrame.new(0, y0 + rise / 2, -hd / 2) * CFrame.Angles(-slope, 0, 0), color, Enum.Material.Slate)

    80→function BuildingKit.fenceLine(parent: Instance, x1: number, z1: number, x2: number, z2: number, groundTop: number)
    81→        local dx, dz = x2 - x1, z2 - z1
    82→        local len = math.sqrt(dx * dx + dz * dz)
    83→        if len < 2 then
    84→                return
    85→        end
    86→        local alongX = math.abs(dx) >= math.abs(dz)
    87→        local size = alongX and Vector3.new(len, 2, 1) or Vector3.new(1, 2, len)
    88→        local mid = Vector3.new((x1 + x2) / 2, groundTop + 3, (z1 + z2) / 2)
    89→        BuildingKit.part(parent, size, CFrame.new(mid), DARKWOOD, Enum.Material.Wood)
    90→        local n = math.max(1, math.floor(len / 40))
    91→        for i = 0, n do
    92→                local t = i / n
    93→                local px = x1 + dx * t
    94→                local pz = z1 + dz * t
    95→                BuildingKit.part(parent, Vector3.new(2, 6, 2), CFrame.new(px, groundTop + 3, pz), DARKWOOD, Enum.Material.Wood)
    96→        end
    97→end

   170→local function hingedDoor(model: Instance, size: Vector3, closedCF: CFrame, hingeLocal: Vector3, openAngle: number): Part
   171→        local leaf = BuildingKit.part(model, size, closedCF, BuildingKit.DARKWOOD, Enum.Material.Wood)
   172→        leaf:SetAttribute("PintuDaun", true)
   173→        local prompt = Instance.new("ProximityPrompt")
   174→        prompt.ActionText = "Buka Pintu"
   175→        prompt.ObjectText = "Pintu"
   176→        prompt.HoldDuration = 0.2
   177→        prompt.MaxActivationDistance = 9
   178→        prompt.RequiresLineOfSight = false
   179→        prompt.Parent = leaf
   180→        local terbuka = false
   181→        local bergerak = false
   182→        local storedClosed: CFrame? = nil
   183→        local function swing(target: CFrame, nextAction: string)
   184→                bergerak = true
   185→                prompt.Enabled = false
   186→                local from = leaf.CFrame
   187→                task.spawn(function()
   188→                        for i = 1, 6 do
   189→                                task.wait(0.04)
   190→                                leaf.CFrame = from:Lerp(target, i / 6)
   191→                        end
   192→                        leaf.CFrame = target
   193→                        prompt.ActionText = nextAction
   194→                        prompt.Enabled = true
   195→                        bergerak = false
   196→                end)
   197→        end
   198→        prompt.Triggered:Connect(function()
   199→                if bergerak then
   200→                        return
   201→                end
   202→                if not terbuka then
   203→                        local base = leaf.CFrame
   204→                        storedClosed = base
   205→                        local hinge = base * CFrame.new(hingeLocal.X, hingeLocal.Y, hingeLocal.Z)
   206→                        local openCF = hinge * CFrame.Angles(0, openAngle, 0) * hinge:Inverse() * base
   207→                        terbuka = true
   208→                        swing(openCF, "Tutup Pintu")
   209→                else
   210→                        terbuka = false
   211→                        swing(storedClosed or leaf.CFrame, "Buka Pintu")
   212→                end
   213→        end)
   214→        return leaf
   215→end
