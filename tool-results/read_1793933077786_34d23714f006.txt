    23→local function fmt(n: number): string
    24→        local s = tostring(math.floor(n))
    25→        local rev = string.reverse(s)
    26→        rev = rev:gsub("(%d%d%d)", "%1.")
    27→        rev = rev:gsub("%.$", "")
    28→        return string.reverse(rev)
    29→end

   414→local function onTap(inputPos: Vector2)
   415→        if dim.Visible then
   416→                return

   419→        local params = RaycastParams.new()

   421→        local char = player.Character

   423→        local result = Workspace:Raycast(ray.Origin, ray.Direction * 400, params)
   424→        if result and result.Instance then
   425→                local plotIdx = result.Instance:GetAttribute("PlotIdx")
   426→                local lahanIdx = result.Instance:GetAttribute("LahanIdx")

   431→                end
