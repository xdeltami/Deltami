   138→   138→function UIController:showPanel(name: string?)
   139→   139→        local TweenService = game:GetService("TweenService")
   140→   140→        for _, n in { "Shop", "Plant", "Inv", "Quiz", "Code", "Help", "Char", "Craft", "Quest", "Trade", "Market", "Kios" } do
   141→   141→                local f = self.gui:FindFirstChild(n)
   142→   142→                if f then
   143→   143→                        (f :: Frame).Visible = (n == name)
   144→   144→                end
   145→   145→        end
   146→   146→        if self.navHolder then
   147→   147→                self.navHolder.Visible = (name == nil)
   148→   148→        end
   149→   149→        if name then
   150→   150→                -- v7 pop-in halus (rasa premium, bukan pop-up kasar)
   151→   151→                local f = self.gui:FindFirstChild(name)
   152→   152→                if f and f:IsA("Frame") then
   153→   153→                        f.Size = UDim2.fromScale(f.Size.X.Scale * 0.94, f.Size.Y.Scale * 0.94)
   154→   154→                        TweenService:Create(f, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
   155→   155→                                Size = UDim2.fromScale(f.Size.X.Scale / 0.94, f.Size.Y.Scale / 0.94),
   156→   156→                        }):Play()
   157→   157→                end
   158→   158→        end
   159→   159→        self.activePanel = name
   160→   160→end
