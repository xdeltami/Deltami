    54→local function btn(props): TextButton
    55→        local b = Util.create("TextButton", {
    56→                Font = Enum.Font.GothamBold,
    57→                TextColor3 = T.Text,
    58→                TextScaled = true,
    59→                AutoButtonColor = true,
    60→                BackgroundColor3 = T.PanelLight,
    61→                BorderSizePixel = 0,
    62→        }, props.parent)
    63→        for k, v in props do
    64→                if k ~= "parent" and k ~= "_radius" then
    65→                        (b :: any)[k] = v
    66→                end
    67→        end
    68→        corner(b, props._radius or 8)
    69→        return b

   123→function UIController:panelFrame(name: string, sizeScale: number, heightScale: number): Frame
   124→        local frame = Util.create("Frame", {
   125→                Name = name,
   126→                AnchorPoint = Vector2.new(0.5, 0.5),
   127→                Position = UDim2.fromScale(0.5, 0.5),
   128→                Size = UDim2.fromScale(sizeScale, heightScale),
   129→                BackgroundColor3 = T.Panel,
   130→                BorderSizePixel = 0,
   131→                Visible = false,
   132→        }, self.gui)
   133→        corner(frame, 14)
   134→        Util.create("UISizeConstraint", { MinSize = Vector2.new(430, 360) }, frame)
   135→        return frame
   136→end

   138→function UIController:showPanel(name: string?)
   139→        local TweenService = game:GetService("TweenService")
   140→        for _, n in { "Shop", "Plant", "Inv", "Quiz", "Code", "Help", "Char", "Craft", "Quest", "Trade", "Market", "Kios" } do
   141→                local f = self.gui:FindFirstChild(n)
   142→                if f then
   143→                        (f :: Frame).Visible = (n == name)
   144→                end
   145→        end
   146→        if self.navHolder then
   147→                self.navHolder.Visible = (name == nil)
   148→        end
   149→        if name then
   150→                -- v7 pop-in halus (rasa premium, bukan pop-up kasar)
   151→                local f = self.gui:FindFirstChild(name)
   152→                if f and f:IsA("Frame") then
   153→                        f.Size = UDim2.fromScale(f.Size.X.Scale * 0.94, f.Size.Y.Scale * 0.94)
   154→                        TweenService:Create(f, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
   155→                                Size = UDim2.fromScale(f.Size.X.Scale / 0.94, f.Size.Y.Scale / 0.94),
   156→                        }):Play()
   157→                end
   158→        end
   159→        self.activePanel = name
   160→end
