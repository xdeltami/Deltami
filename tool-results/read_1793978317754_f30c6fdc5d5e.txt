   146→local function jawabSoal(gui)
   147→        local soalTeks = findSoalTeks(gui)
   148→        if not soalTeks then
   149→                return false, "soal hilang"
   150→        end
   151→        local teks = soalTeks.Text
   152→        local a, op, b = string.match(teks, "^(%d+) (.+) (%d+) = %?$")
   153→        if not a then
   154→                return false, "soal tak terbaca: " .. teks
   155→        end
   156→        local x, y = tonumber(a), tonumber(b)
   157→        local hasil
   158→        if op == "+" then
   159→                hasil = x + y
   160→        elseif op == "-" then
   161→                hasil = x - y
   162→        elseif op == "*" or op == "x" or op == "×" then
   163→                hasil = x * y
   164→        elseif op == "/" or op == "÷" then
   165→                hasil = x / y
   166→        else
   167→                return false, "operator tak dikenal: " .. tostring(op)
   168→        end
   169→        local box = findTextBoxByPlaceholder(gui, "Tulis jawaban angka")
   170→        if not box then
   171→                return false, "kotak jawaban hilang"
   172→        end
   173→        box.Text = tostring(hasil)
   174→        local btn = findGui(gui, "TextButton", "Jawab")
   175→        if not btn then
   176→                return false, "tombol jawab hilang"
   177→        end
   178→        btn.Activated:Fire()
   179→        return true, tostring(hasil)
   180→end
