    20→local function fmt(n: number): string
    21→        local s = tostring(math.floor(n))
    22→        local rev = string.reverse(s)
    23→        rev = rev:gsub("(%d%d%d)", "%1.")
    24→        rev = rev:gsub("%.$", "")
    25→        return string.reverse(rev)
    26→end

   430→local function onTap(inputPos: Vector2)
   431→        if dim.Visible then
   432→                return
   433→        end
   434→        local ray = camera:ViewportPointToRay(inputPos.X, inputPos.Y)
   435→        local params = RaycastParams.new()
   436→        params.FilterType = Enum.RaycastFilterType.Exclude
   437→        local char = player.Character
   438→        params.FilterDescendantsInstances = char and { char } or {}
   439→        local result = Workspace:Raycast(ray.Origin, ray.Direction * 400, params)
   440→        if result and result.Instance then
   441→                local plotIdx = result.Instance:GetAttribute("PlotIdx")
   442→                local lahanIdx = result.Instance:GetAttribute("LahanIdx")
   443→                if plotIdx ~= nil and lahanIdx ~= nil then
   444→                        plotCtx = { lahan = lahanIdx, idx = plotIdx }
   445→                        openPanel("plot")
   446→                        refreshPlotPanel()
   447→                end
   448→        end
   449→end
