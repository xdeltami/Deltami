    34→local function fmt(n: number): string
    35→        local s = tostring(math.floor(n))
    36→        local rev = string.reverse(s)
    37→        rev = rev:gsub("(%d%d%d)", "%1.")
    38→        rev = rev:gsub("%.$", "")
    39→        return string.reverse(rev)
    40→end
